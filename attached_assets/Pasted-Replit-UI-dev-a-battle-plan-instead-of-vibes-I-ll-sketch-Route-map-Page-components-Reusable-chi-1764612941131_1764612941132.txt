1️⃣ Route Map (React Router / Next.js style)

Assuming a standard SPA with React Router (if you’re on Next.js, these become page files instead of route entries):

/dashboard
→ High-level overview, shortcut tiles to upload & reports.

/reports
→ ReportsListPage – list all parsed credit reports for the logged-in user.

/reports/:reportId
→ ReportDetailPage – show report metadata, issue list, and letter generator.

/reports/upload
→ ReportUploadPage – drag-and-drop upload → calls /api/credit-reports/upload.

/letters (optional later)
→ LettersHistoryPage – list all generated letters with download links.

Pseudo-structure:

// src/pages/ReportDetailPage.tsx
import { useParams } from 'react-router-dom';
import { useCreditReport } from '../hooks/useCreditReport';
import { useViolations } from '../hooks/useViolations';
import { useGenerateLetter } from '../hooks/useGenerateLetter';

export function ReportDetailPage() {
  const { reportId } = useParams<{ reportId: string }>();

  const { data: report, isLoading: reportLoading, error: reportError } =
    useCreditReport(reportId!);
  const { data: violations, isLoading: vLoading, error: vError } =
    useViolations(reportId!);

  const {
    generateLetter,
    isGenerating,
    lastGenerated
  } = useGenerateLetter();

  const [selectedViolationIds, setSelectedViolationIds] = React.useState<string[]>([]);
  const [recipientType, setRecipientType] = React.useState<'BUREAU' | 'COLLECTOR'>('BUREAU');
  const [targetBureau, setTargetBureau] = React.useState<'EXPERIAN' | 'EQUIFAX' | 'TRANSUNION'>('EXPERIAN');
  const [targetCollector, setTargetCollector] = React.useState<string>('');

  const handleGenerate = async () => {
    if (!reportId || selectedViolationIds.length === 0) return;

    const payload =
      recipientType === 'BUREAU'
        ? {
            reportId,
            violationIds: selectedViolationIds,
            round: 1,
            recipientType: 'BUREAU',
            target: targetBureau
          }
        : {
            reportId,
            violationIds: selectedViolationIds,
            round: 1,
            recipientType: 'COLLECTOR',
            target: targetCollector
          };

    await generateLetter(payload);
  };

  if (reportLoading || vLoading) return <p>Loading...</p>;
  if (reportError) return <p className="error">{reportError}</p>;
  if (!report) return <p>Report not found.</p>;

  return (
    <div className="page">
      <h1>Report Details</h1>

      <ReportSummaryCard report={report} />

      <div className="layout-two-col">
        <div className="col col-main">
          <ViolationsFilterBar
            violations={violations || []}
            onFilterChange={/* optional filters */}
          />

          <ViolationsTable
            violations={violations || []}
            selectedIds={selectedViolationIds}
            onSelectionChange={setSelectedViolationIds}
          />
        </div>

        <div className="col col-side">
          <LetterGenerationPanel
            violations={violations || []}
            selectedIds={selectedViolationIds}
            recipientType={recipientType}
            onRecipientTypeChange={setRecipientType}
            targetBureau={targetBureau}
            onTargetBureauChange={setTargetBureau}
            targetCollector={targetCollector}
            onTargetCollectorChange={setTargetCollector}
            onGenerate={handleGenerate}
            isGenerating={isGenerating}
            lastGenerated={lastGenerated}
          />
        </div>
      </div>
    </div>
  );
}


Key behavior:

<ViolationsTable /> shows issues grouped by category or with chips.

Selection drives which issues go into /api/dispute-letters/generate.

<LetterGenerationPanel /> computes:

Distinct collectors from selected issues (issue.collectorName).

Bureaus from issue.relatedAccount.bureauScope.

4️⃣ Hooks Layer (Data Access)

You can implement these with fetch or axios. I’ll show a generic shape with fetch and React Query-ish patterns (you can adapt).

4.1 useCreditReports
// src/hooks/useCreditReports.ts
import { useEffect, useState } from 'react';
import { CreditReportSummary } from '../types/api';

export function useCreditReports() {
  const [data, setData] = useState<CreditReportSummary[] | null>(null);
  const [isLoading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    let cancelled = false;
    setLoading(true);
    fetch('/api/credit-reports')
      .then(async (res) => {
        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        return res.json();
      })
      .then((json) => {
        if (!cancelled) setData(json.reports);
      })
      .catch((err) => {
        if (!cancelled) setError(err.message || 'Failed to load reports');
      })
      .finally(() => {
        if (!cancelled) setLoading(false);
      });

    return () => {
      cancelled = true;
    };
  }, []);

  return { data, isLoading, error };
}


Backend expectation:

GET /api/credit-reports → { reports: CreditReportSummary[] } where CreditReportSummary includes:

id

meta.reportDate

bureau

provider

violationsCount

4.4 useUploadCreditReport
// src/hooks/useUploadCreditReport.ts
import { useState } from 'react';

type UploadOptions = {
  bureau?: 'TRI' | 'EXPERIAN' | 'EQUIFAX' | 'TRANSUNION';
};

type UploadResult = {
  reportId: string;
  issuesCount: number;
};

export function useUploadCreditReport() {
  const [isUploading, setUploading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  async function upload(file: File, options: UploadOptions): Promise<UploadResult | null> {
    setUploading(true);
    setError(null);

    try {
      // Step 1: upload file to storage (if you have a dedicated file endpoint)
      const formData = new FormData();
      formData.append('file', file);

      const uploadRes = await fetch('/api/files/upload', {
        method: 'POST',
        body: formData
      });

      if (!uploadRes.ok) throw new Error(`File upload failed: ${uploadRes.status}`);
      const { filePath } = await uploadRes.json();

      // Step 2: tell backend to parse + analyze
      const res = await fetch('/api/credit-reports/upload', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          filePath,
          bureau: options.bureau || 'TRI'
        })
      });

      if (!res.ok) throw new Error(`Upload/parse failed: ${res.status}`);
      const json = await res.json();
      return { reportId: json.reportId, issuesCount: json.issuesCount };
    } catch (err: any) {
      setError(err.message || 'Upload failed');
      return null;
    } finally {
      setUploading(false);
    }
  }

  return { upload, isUploading, error };
}


Backend expectation:

POST /api/files/upload → { filePath: string }

POST /api/credit-reports/upload → { ok: true, reportId, issuesCount }

4.5 useGenerateLetter
// src/hooks/useGenerateLetter.ts
import { useState } from 'react';

type GenerateLetterPayload = {
  reportId: string;
  violationIds: string[];
  round: 1 | 2 | 3;
  recipientType: 'BUREAU' | 'COLLECTOR';
  target: string; // 'EXPERIAN' | 'EQUIFAX' | 'TRANSUNION' or collector name
};

type GenerateLetterResponse = {
  ok: boolean;
  letterId: string;
  pdfUrl: string;
  recipientType: 'BUREAU' | 'COLLECTOR';
  recipient: string;
  round: number;
};

export function useGenerateLetter() {
  const [isGenerating, setGenerating] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [lastGenerated, setLastGenerated] = useState<GenerateLetterResponse | null>(null);

  async function generateLetter(payload: GenerateLetterPayload) {
    setGenerating(true);
    setError(null);

    try {
      const res = await fetch('/api/dispute-letters/generate', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });

      if (!res.ok) throw new Error(`HTTP ${res.status}`);
      const json = (await res.json()) as GenerateLetterResponse;
      setLastGenerated(json);
      return json;
    } catch (err: any) {
      setError(err.message || 'Failed to generate letter');
      return null;
    } finally {
      setGenerating(false);
    }
  }

  return { generateLetter, isGenerating, error, lastGenerated };
}


Backend expectation:
POST /api/dispute-letters/generate → { ok, letterId, pdfUrl, recipientType, recipient, round }.
