Here’s a concrete generateAllDisputeLetters module that:

Groups IssueFlags per tradeline / collection.

Joins them with AiIssueExplanation.fullText.

Produces:

3 bureau letters (Experian, Equifax, TransUnion), and

furnisher letters (creditors + collectors) ready to push into your HTML→PDF layer.

Adjust import paths and field names to your actual types.

// src/server/services/letters/generateAllDisputeLetters.ts

import {
  CreditReport,
  Tradeline,
  CollectionAccount,
  PublicRecord,
  BureauCode
} from '../types/credit'; // adjust path
import { IssueFlag, AiIssueExplanation } from '../rules/ruleImplementations'; // adjust path

export interface ConsumerProfile {
  fullName: string;
  addressLine1: string;
  addressLine2?: string;
  city: string;
  state: string;
  postalCode: string;
  phone?: string;
  email?: string;
  ssnLast4?: string;
  dob?: string; // MM/DD/YYYY
}

export interface BureauDisputeLetter {
  bureau: BureauCode;
  toAddressLines: string[];
  subject: string;
  body: string; // full letter text, ready for HTML/PDF
}

export interface FurnisherDisputeLetter {
  furnisherName: string;
  furnisherType: 'CREDITOR' | 'COLLECTOR';
  toAddressLines: string[]; // you can later auto-fill or leave as placeholders
  subject: string;
  body: string;
  relatedTradelineIds: string[];
  relatedCollectionIds: string[];
}

export interface DisputeLetterBundle {
  bureauLetters: BureauDisputeLetter[];
  furnisherLetters: FurnisherDisputeLetter[];
}

// ---------------------------------------------------------------------
// Public entry
// ---------------------------------------------------------------------

export function generateAllDisputeLetters(params: {
  report: CreditReport;
  consumer: ConsumerProfile;
  issues: IssueFlag[];
  aiExplanations: AiIssueExplanation[];
}): DisputeLetterBundle {
  const { report, consumer, issues, aiExplanations } = params;

  const explanationMap = buildExplanationMap(aiExplanations);

  // Group issues per account
  const {
    tradelineIssuesById,
    collectionIssuesById,
    publicRecordIssuesById
  } = groupIssuesByAccount(issues, explanationMap);

  // 1) Bureau dispute letters (FCRA + Metro 2 centered)
  const bureauLetters: BureauDisputeLetter[] = [
    buildBureauLetter('EXPERIAN', report, consumer, tradelineIssuesById, collectionIssuesById, publicRecordIssuesById),
    buildBureauLetter('EQUIFAX', report, consumer, tradelineIssuesById, collectionIssuesById, publicRecordIssuesById),
    buildBureauLetter('TRANSUNION', report, consumer, tradelineIssuesById, collectionIssuesById, publicRecordIssuesById)
  ];

  // 2) Furnisher letters (creditors + collectors, FCRA + Metro 2 + FDCPA)
  const furnisherLetters = buildFurnisherLetters(
    report,
    consumer,
    tradelineIssuesById,
    collectionIssuesById
  );

  return { bureauLetters, furnisherLetters };
}

// ---------------------------------------------------------------------
// Internal helpers
// ---------------------------------------------------------------------

function buildExplanationMap(aiExplanations: AiIssueExplanation[]): Map<string, AiIssueExplanation> {
  const map = new Map<string, AiIssueExplanation>();
  for (const exp of aiExplanations) {
    map.set(exp.issueId, exp);
  }
  return map;
}

interface IssueWithExplanation {
  issue: IssueFlag;
  explanation?: AiIssueExplanation;
}

function groupIssuesByAccount(
  issues: IssueFlag[],
  explanationMap: Map<string, AiIssueExplanation>
): {
  tradelineIssuesById: Map<string, IssueWithExplanation[]>;
  collectionIssuesById: Map<string, IssueWithExplanation[]>;
  publicRecordIssuesById: Map<string, IssueWithExplanation[]>;
} {
  const tradelineIssuesById = new Map<string, IssueWithExplanation[]>();
  const collectionIssuesById = new Map<string, IssueWithExplanation[]>();
  const publicRecordIssuesById = new Map<string, IssueWithExplanation[]>();

  const push = (
    map: Map<string, IssueWithExplanation[]>,
    id: string | undefined,
    issue: IssueFlag
  ) => {
    if (!id) return;
    if (!map.has(id)) map.set(id, []);
    map.get(id)!.push({
      issue,
      explanation: explanationMap.get(issue.id)
    });
  };

  for (const issue of issues) {
    const ra = issue.relatedAccount;
    if (ra.type === 'TRADELINE') {
      push(tradelineIssuesById, ra.tradelineId, issue);
    } else if (ra.type === 'COLLECTION') {
      push(collectionIssuesById, ra.collectionId, issue);
    } else if (ra.type === 'PUBLIC_RECORD') {
      push(publicRecordIssuesById, ra.publicRecordId, issue);
    }
  }

  return { tradelineIssuesById, collectionIssuesById, publicRecordIssuesById };
}

const BUREAU_ADDRESSES: Record<BureauCode, string[]> = {
  EXPERIAN: ['Experian', 'P.O. Box 4500', 'Allen, TX 75013'],
  EQUIFAX: ['Equifax Information Services LLC', 'P.O. Box 740256', 'Atlanta, GA 30374-0256'],
  TRANSUNION: ['TransUnion LLC', 'Consumer Dispute Center', 'P.O. Box 2000', 'Chester, PA 19016']
};

function buildBureauLetter(
  bureau: BureauCode,
  report: CreditReport,
  consumer: ConsumerProfile,
  tradelineIssuesById: Map<string, IssueWithExplanation[]>,
  collectionIssuesById: Map<string, IssueWithExplanation[]>,
  publicRecordIssuesById: Map<string, IssueWithExplanation[]>
): BureauDisputeLetter {
  const toAddressLines = BUREAU_ADDRESSES[bureau];
  const subject = 'Request for Investigation of Inaccurate Credit Reporting';

  const reportDate = report.meta?.reportDate ?? '[REPORT DATE]';

  // Filter accounts that actually show that bureau
  const tradelinesForBureau = report.tradelines.filter((t) =>
    t.bureaus.includes(bureau)
  );
  const collectionsForBureau = report.collections.filter((c) =>
    c.bureaus.includes(bureau)
  );
  const publicRecordsForBureau = report.publicRecords.filter((p) =>
    p.bureaus.includes(bureau)
  );

  const headerBlock = buildConsumerHeader(consumer);

  const intro = [
    'Re: Request for Investigation of Inaccurate Credit Reporting',
    `Name: ${consumer.fullName}`,
    consumer.ssnLast4 ? `SSN (last 4): XXX-XX-${consumer.ssnLast4}` : undefined,
    consumer.dob ? `DOB: ${consumer.dob}` : undefined,
    `Credit Report Date: ${reportDate}`
  ]
    .filter(Boolean)
    .join('\n');

  const introParagraph =
    'I am writing to dispute several inaccurate, incomplete, or misleading items appearing on my credit file with your bureau. ' +
    'This letter is a formal request for reinvestigation under the Fair Credit Reporting Act (FCRA), 15 U.S.C. § 1681i. ' +
    'In some cases, the information also appears to deviate from industry Metro 2 reporting standards, which may cause the data to be interpreted in a way that is unfairly harmful.';

  const sections: string[] = [];

  // Tradeline sections
  for (const tl of tradelinesForBureau) {
    const issueList = tradelineIssuesById.get(tl.id) || [];
    if (!issueList.length) continue;

    const header = `--------------------------------------------------\n${labelTradeline(tl)}\n--------------------------------------------------`;
    const basicInfo = [
      `Creditor: ${tl.creditorName}`,
      `Account Type: ${tl.type}`,
      tl.accountNumberLast4 ? `Account Ending In: ${tl.accountNumberLast4}` : undefined
    ]
      .filter(Boolean)
      .join('\n');

    const reasons = buildReasonsParagraphs(issueList);

    sections.push([header, basicInfo, 'Dispute Reasons:', reasons].join('\n\n'));
  }

  // Collection sections
  for (const col of collectionsForBureau) {
    const issueList = collectionIssuesById.get(col.id) || [];
    if (!issueList.length) continue;

    const header = `--------------------------------------------------\n${labelCollection(col)}\n--------------------------------------------------`;
    const basicInfo = [
      `Collector: ${col.collectorName}`,
      col.originalCreditorName ? `Original Creditor: ${col.originalCreditorName}` : undefined,
      `Account Status: ${col.status}`,
      col.balance != null ? `Reported Balance: $${col.balance.toFixed(2)}` : undefined
    ]
      .filter(Boolean)
      .join('\n');

    const reasons = buildReasonsParagraphs(
      // For bureaus, we primarily focus on FCRA + METRO2, but FDCPA reasoning is still ok
      issueList.filter((iw) => iw.issue.category === 'FCRA' || iw.issue.category === 'METRO2' || iw.issue.category === 'FDCPA')
    );

    sections.push([header, basicInfo, 'Dispute Reasons:', reasons].join('\n\n'));
  }

  // Public record sections (if you want them)
  for (const pr of publicRecordsForBureau) {
    const issueList = publicRecordIssuesById.get(pr.id) || [];
    if (!issueList.length) continue;

    const header = `--------------------------------------------------\nPublic Record – ${pr.type}\n--------------------------------------------------`;
    const basicInfo = [
      `Court: ${pr.courtName || '[Court not clear on report]'}`,
      pr.amount != null ? `Reported Amount: $${pr.amount.toFixed(2)}` : undefined,
      pr.filedDate ? `Filed Date: ${pr.filedDate}` : undefined
    ]
      .filter(Boolean)
      .join('\n');

    const reasons = buildReasonsParagraphs(issueList);

    sections.push([header, basicInfo, 'Dispute Reasons:', reasons].join('\n\n'));
  }

  const closing =
    'Pursuant to 15 U.S.C. § 1681i, please conduct a reasonable reinvestigation of each disputed item listed above, ' +
    'including contacting each furnisher of information. For any item that cannot be verified as fully accurate and properly coded, ' +
    'please delete it from my credit file with your bureau. After completing your reinvestigation, kindly send me an updated copy of my credit report ' +
    'and a description of the procedures used, including the names and addresses of any furnishers you contacted.\n\n' +
    'I have enclosed copies of my identification, proof of address, and a copy of my credit report with the disputed items highlighted.\n\n' +
    'Sincerely,\n\n' +
    `${consumer.fullName}\n${consumer.addressLine1}${consumer.addressLine2 ? '\n' + consumer.addressLine2 : ''}\n${consumer.city}, ${consumer.state} ${consumer.postalCode}`;

  const body = [
    headerBlock,
    '',
    intro,
    '',
    introParagraph,
    '',
    sections.length
      ? sections.join('\n\n')
      : '(No specific tradelines or collections identified for this bureau. If this happens, check grouping logic.)',
    '',
    closing
  ].join('\n');

  return {
    bureau,
    toAddressLines,
    subject,
    body
  };
}

function buildConsumerHeader(consumer: ConsumerProfile): string {
  return [
    consumer.fullName,
    consumer.addressLine1,
    consumer.addressLine2,
    `${consumer.city}, ${consumer.state} ${consumer.postalCode}`,
    consumer.phone ? `Phone: ${consumer.phone}` : undefined,
    consumer.email ? `Email: ${consumer.email}` : undefined,
    '[Date]'
  ]
    .filter(Boolean)
    .join('\n');
}

// Furnisher letters (creditors + collectors)
// ---------------------------------------------------------------------

function buildFurnisherLetters(
  report: CreditReport,
  consumer: ConsumerProfile,
  tradelineIssuesById: Map<string, IssueWithExplanation[]>,
  collectionIssuesById: Map<string, IssueWithExplanation[]>
): FurnisherDisputeLetter[] {
  const letters: FurnisherDisputeLetter[] = [];

  // Group creditor tradeline issues by creditorName
  const creditorMap = new Map<
    string,
    { creditorName: string; tradelineIds: string[]; issues: IssueWithExplanation[] }
  >();

  for (const tl of report.tradelines) {
    const issues = tradelineIssuesById.get(tl.id) || [];
    if (!issues.length) continue;

    const key = (tl.creditorName || '').trim();
    if (!key) continue;

    if (!creditorMap.has(key)) {
      creditorMap.set(key, { creditorName: key, tradelineIds: [], issues: [] });
    }
    const entry = creditorMap.get(key)!;
    entry.tradelineIds.push(tl.id);
    entry.issues.push(...issues);
  }

  for (const [, entry] of creditorMap) {
    letters.push(buildCreditorLetter(entry, report, consumer));
  }

  // Group collector issues by collectorName
  const collectorMap = new Map<
    string,
    {
      collectorName: string;
      collectionIds: string[];
      issues: IssueWithExplanation[];
      originalCreditorNames: Set<string>;
    }
  >();

  for (const col of report.collections) {
    const issues = collectionIssuesById.get(col.id) || [];
    if (!issues.length) continue;

    const key = (col.collectorName || '').trim();
    if (!key) continue;

    if (!collectorMap.has(key)) {
      collectorMap.set(key, {
        collectorName: key,
        collectionIds: [],
        issues: [],
        originalCreditorNames: new Set()
      });
    }
    const entry = collectorMap.get(key)!;
    entry.collectionIds.push(col.id);
    entry.issues.push(...issues);
    if (col.originalCreditorName) entry.originalCreditorNames.add(col.originalCreditorName);
  }

  for (const [, entry] of collectorMap) {
    letters.push(buildCollectorLetter(entry, report, consumer));
  }

  return letters;
}

function buildCreditorLetter(
  entry: {
    creditorName: string;
    tradelineIds: string[];
    issues: IssueWithExplanation[];
  },
  report: CreditReport,
  consumer: ConsumerProfile
): FurnisherDisputeLetter {
  const creditorName = entry.creditorName;
  const header = buildConsumerHeader(consumer);

  const subject = 'Direct Dispute of Inaccurate Credit Reporting – FCRA § 623';

  const tradelines = report.tradelines.filter((t) => entry.tradelineIds.includes(t.id));

  const basicIntro =
    `Re: Direct Dispute of Credit Reporting for Accounts with ${creditorName}\n\n` +
    'This letter is a formal direct dispute under the Fair Credit Reporting Act (FCRA), including 15 U.S.C. § 1681s-2, regarding the accuracy and completeness of the information ' +
    `that ${creditorName} is furnishing to the consumer reporting agencies. The following accounts and issues appear on my recent credit report and are disputed as inaccurate, incomplete, or misleading.`;

  const sections: string[] = [];

  for (const tl of tradelines) {
    const issueList = entry.issues.filter(
      (iw) => iw.issue.relatedAccount.tradelineId === tl.id
    );
    if (!issueList.length) continue;

    const headerLine = labelTradeline(tl);
    const basicInfo = [
      `Creditor: ${tl.creditorName}`,
      tl.accountNumberLast4 ? `Account Ending In: ${tl.accountNumberLast4}` : undefined
    ]
      .filter(Boolean)
      .join('\n');

    const reasons = buildReasonsParagraphs(issueList);

    sections.push([headerLine, basicInfo, 'Dispute Reasons:', reasons].join('\n\n'));
  }

  const closing =
    'Please conduct a reasonable investigation of each disputed item, including review of your own account records and servicing notes. ' +
    'After your investigation, please correct or delete any information that cannot be verified as fully accurate and properly coded, ' +
    'and report all corrections to every consumer reporting agency to which you furnished the information.\n\n' +
    'Please provide written confirmation of the results of your investigation and copies of any documentation you relied upon.\n\n' +
    'Sincerely,\n\n' +
    `${consumer.fullName}\n${consumer.addressLine1}${consumer.addressLine2 ? '\n' + consumer.addressLine2 : ''}\n${consumer.city}, ${consumer.state} ${consumer.postalCode}`;

  const body = [header, '', '[Creditor Disputes / Correspondence Address]', '', basicIntro, '', sections.join('\n\n'), '', closing].join('\n');

  return {
    furnisherName: creditorName,
    furnisherType: 'CREDITOR',
    toAddressLines: [creditorName, '[Address Line 1]', '[Address Line 2]'],
    subject,
    body,
    relatedTradelineIds: entry.tradelineIds,
    relatedCollectionIds: []
  };
}

function buildCollectorLetter(
  entry: {
    collectorName: string;
    collectionIds: string[];
    issues: IssueWithExplanation[];
    originalCreditorNames: Set<string>;
  },
  report: CreditReport,
  consumer: ConsumerProfile
): FurnisherDisputeLetter {
  const collectorName = entry.collectorName;
  const header = buildConsumerHeader(consumer);

  const subject =
    'Dispute and Request for Validation of Reported Collection Accounts – FCRA & FDCPA';

  const collections = report.collections.filter((c) =>
    entry.collectionIds.includes(c.id)
  );

  const ocList = Array.from(entry.originalCreditorNames).join(', ') || 'unknown original creditor(s)';

  const basicIntro =
    `Re: Dispute and Validation Request for Collection Accounts Reported by ${collectorName}\n\n` +
    `This letter disputes the accuracy and completeness of collection accounts that ${collectorName} is reporting in connection with debts allegedly originating with ${ocList}. ` +
    'This dispute is made under the Fair Credit Reporting Act (FCRA) regarding the accuracy of data furnished to the credit bureaus, and under the Fair Debt Collection Practices Act (FDCPA), ' +
    'including 15 U.S.C. § 1692g and § 1692e, regarding the accuracy, validation, and characterization of these alleged debts.';

  const sections: string[] = [];

  for (const col of collections) {
    const issueList = entry.issues.filter(
      (iw) => iw.issue.relatedAccount.collectionId === col.id
    );
    if (!issueList.length) continue;

    const headerLine = labelCollection(col);
    const basicInfo = [
      `Collector: ${col.collectorName}`,
      col.originalCreditorName ? `Original Creditor: ${col.originalCreditorName}` : undefined,
      col.balance != null ? `Reported Balance: $${col.balance.toFixed(2)}` : undefined
    ]
      .filter(Boolean)
      .join('\n');

    const reasons = buildReasonsParagraphs(issueList);

    sections.push([headerLine, basicInfo, 'Dispute / Validation Reasons:', reasons].join('\n\n'));
  }

  const closing =
    'Please provide proper validation of each alleged debt, including chain of title from the original creditor, a complete itemization of the balance, and copies of any agreements that allegedly created the obligation. ' +
    'Until full validation is provided, any continued collection activity or credit reporting of disputed, unvalidated accounts may violate the FDCPA.\n\n' +
    'In addition, please correct or delete any inaccurate, duplicated, re-aged, or improperly coded information you have furnished to the consumer reporting agencies, ' +
    'and confirm those corrections in writing.\n\n' +
    'Sincerely,\n\n' +
    `${consumer.fullName}\n${consumer.addressLine1}${consumer.addressLine2 ? '\n' + consumer.addressLine2 : ''}\n${consumer.city}, ${consumer.state} ${consumer.postalCode}`;

  const body = [header, '', '[Collections / Correspondence Address]', '', basicIntro, '', sections.join('\n\n'), '', closing].join('\n');

  return {
    furnisherName: collectorName,
    furnisherType: 'COLLECTOR',
    toAddressLines: [collectorName, '[Address Line 1]', '[Address Line 2]'],
    subject,
    body,
    relatedTradelineIds: [],
    relatedCollectionIds: entry.collectionIds
  };
}

// ---------------------------------------------------------------------
// Shared formatting helpers
// ---------------------------------------------------------------------

function labelTradeline(tl: Tradeline): string {
  const base = tl.creditorName || 'Unknown Creditor';
  const tail = tl.accountNumberLast4 ? ` – Account Ending in ${tl.accountNumberLast4}` : '';
  return `${base}${tail}`;
}

function labelCollection(col: CollectionAccount): string {
  const base = col.collectorName || 'Unknown Collection Agency';
  const oc = col.originalCreditorName ? ` (Original Creditor: ${col.originalCreditorName})` : '';
  return `${base}${oc}`;
}

function buildReasonsParagraphs(issueList: IssueWithExplanation[]): string {
  if (!issueList.length) {
    return '(No issue explanations attached – check rules engine wiring.)';
  }

  const paragraphs: string[] = [];

  // Deduplicate by ruleId + explanation text to avoid spam
  const seen = new Set<string>();

  for (const { issue, explanation } of issueList) {
    const text = (explanation?.fullText || explanation?.summary || issue.description).trim();
    if (!text) continue;
    const key = `${issue.ruleId}|${text}`;
    if (seen.has(key)) continue;
    seen.add(key);
    // Format as bullet-style paragraph for readability
    paragraphs.push(`- ${text}`);
  }

  if (!paragraphs.length) {
    return '(No issue explanations attached – check AI explanation generator.)';
  }

  return paragraphs.join('\n');
}
