✅ 1. 🔧 Plan Selector UI + Logic
📄 main.py (Add this to existing FastAPI app)
from fastapi import Form, Cookie
from fastapi.responses import RedirectResponse

import json

def load_user(email):
    with open("users.json") as f:
        users = json.load(f)
    return users.get(email)

def save_user(email, user_data):
    with open("users.json") as f:
        users = json.load(f)
    users[email] = user_data
    with open("users.json", "w") as f:
        json.dump(users, f, indent=2)

@app.get("/select-plan", response_class=HTMLResponse)
def select_plan(request: Request, user_email: str = Cookie(None)):
    if not user_email:
        return RedirectResponse("/login")
    return templates.TemplateResponse("select_plan.html", {"request": request})

@app.post("/select-plan")
def update_plan(plan: str = Form(...), user_email: str = Cookie(None)):
    if not user_email:
        return RedirectResponse("/login")
    user = load_user(user_email)
    user["plan"] = plan
    save_user(user_email, user)
    return RedirectResponse("/dashboard", status_code=302)

🤖 2. GPT AI Assistant Integration (Plan-Aware)
📄 ai_assistant.py
import openai
import os
from dotenv import load_dotenv

load_dotenv()

openai.api_key = os.getenv("OPENAI_API_KEY")

def get_ai_reply(user_input, plan):
    system_prompt = {
        "diy": """You are a credit repair coach for DIY users.
You help them understand dispute letters and guide them to send letters manually. Do NOT mention automated tools, mailing services, or CRM.""",
        "pro": """You are an AI assistant for a full-service automated credit repair app.
You track dispute letter status, mailing confirmations, and CRM updates. Help the user understand what's happening behind the scenes."""
    }

    chat = openai.ChatCompletion.create(
        model="gpt-4",
        messages=[
            {"role": "system", "content": system_prompt[plan]},
            {"role": "user", "content": user_input}
        ]
    )
    return chat.choices[0].message.content
