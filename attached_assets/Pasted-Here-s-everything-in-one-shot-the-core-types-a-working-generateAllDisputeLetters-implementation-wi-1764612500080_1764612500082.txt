B. Minimal generateAllDisputeLetters Implementation

This function:

Takes a CreditReport, IssueFlag[], AiIssueExplanation[], and UserInfo.

Groups issues:

By bureau (FCRA / Metro 2 angles)

By collector (FDCPA + furnisher disputes)

Assembles letters that already include:

FCRA “maximum possible accuracy” language

Metro 2 “non-compliant coding / misleading reporting” language

FDCPA validation + cease-collection language

Create server/services/enhancedDisputeLetters.ts:

// server/services/enhancedDisputeLetters.ts
import {
  BureauCode,
  CreditReport,
  IssueFlag,
  AiIssueExplanation,
  UserInfo,
  GeneratedLetter,
  StatuteReference
} from '../types/credit';

// Utility: default bureau mailing names (you can still override later)
const BUREAU_LEGAL_NAMES: Record<Exclude<BureauCode, 'TRI'>, string> = {
  EXPERIAN: 'Experian',
  EQUIFAX: 'Equifax Information Services LLC',
  TRANSUNION: 'TransUnion LLC'
};

interface GenerateLettersInput {
  report: CreditReport;
  issues: IssueFlag[];
  aiExplanations: AiIssueExplanation[];
  userInfo: UserInfo;
  round: 1 | 2 | 3;
}

interface GenerateLettersOutput {
  bureauLetters: Partial<Record<Exclude<BureauCode, 'TRI'>, GeneratedLetter>>;
  collectorLetters: Record<string, GeneratedLetter>;
}

export async function generateAllDisputeLetters(
  input: GenerateLettersInput
): Promise<GenerateLettersOutput> {
  const { report, issues, aiExplanations, userInfo, round } = input;

  // 1️⃣ Map explanations by (ruleId + issueId) so we can attach narratives
  const explanationIndex = buildExplanationIndex(aiExplanations);

  // 2️⃣ Group issues by bureau and by collector
  const issuesByBureau: Partial<Record<Exclude<BureauCode, 'TRI'>, IssueFlag[]>> = {
    EXPERIAN: [],
    EQUIFAX: [],
    TRANSUNION: []
  };

  const issuesByCollector: Record<string, IssueFlag[]> = {};

  for (const issue of issues) {
    const bureaus = issue.relatedAccount.bureauScope || [report.bureau].filter(
      (b): b is Exclude<BureauCode, 'TRI'> =>
        b === 'EXPERIAN' || b === 'EQUIFAX' || b === 'TRANSUNION'
    );

    for (const b of bureaus) {
      if (!issuesByBureau[b]) issuesByBureau[b] = [];
      issuesByBureau[b]!.push(issue);
    }

    const collectorName =
      issue.collectorName || issue.relatedAccount.collectorName;
    if (collectorName) {
      const normalized = collectorName.trim();
      if (!issuesByCollector[normalized]) issuesByCollector[normalized] = [];
      issuesByCollector[normalized].push(issue);
    }
  }

  // 3️⃣ Build bureau letters
  const bureauLetters: Partial<Record<Exclude<BureauCode, 'TRI'>, GeneratedLetter>> = {};

  (['EXPERIAN', 'EQUIFAX', 'TRANSUNION'] as const).forEach((bureau) => {
    const bureauIssues = issuesByBureau[bureau] || [];
    if (!bureauIssues.length) return; // no letter if no issues

    const letter = buildBureauLetter({
      bureau,
      bureauIssues,
      explanationIndex,
      userInfo,
      report,
      round
    });

    bureauLetters[bureau] = letter;
  });

  // 4️⃣ Build collector letters
  const collectorLetters: Record<string, GeneratedLetter> = {};
  for (const collectorName of Object.keys(issuesByCollector)) {
    const collectorIssues = issuesByCollector[collectorName];
    const letter = buildCollectorLetter({
      collectorName,
      collectorIssues,
      explanationIndex,
      userInfo,
      report,
      round
    });
    collectorLetters[collectorName] = letter;
  }

  return { bureauLetters, collectorLetters };
}

// --- Explanation index helper ---

type ExplanationKey = string;

function makeExplanationKey(ruleId: string, issueId?: string) {
  return issueId ? `${ruleId}::${issueId}` : ruleId;
}

function buildExplanationIndex(
  explanations: AiIssueExplanation[]
): Record<ExplanationKey, AiIssueExplanation> {
  const index: Record<string, AiIssueExplanation> = {};
  for (const exp of explanations) {
    index[makeExplanationKey(exp.ruleId, exp.issueId)] = exp;
    // also index by ruleId alone as fallback
    if (!index[exp.ruleId]) {
      index[exp.ruleId] = exp;
    }
  }
  return index;
}

// --- Bureau letter builder (FCRA + Metro 2) ---

interface BureauLetterParams {
  bureau: Exclude<BureauCode, 'TRI'>;
  bureauIssues: IssueFlag[];
  explanationIndex: Record<ExplanationKey, AiIssueExplanation>;
  userInfo: UserInfo;
  report: CreditReport;
  round: 1 | 2 | 3;
}

function buildBureauLetter(params: BureauLetterParams): GeneratedLetter {
  const { bureau, bureauIssues, explanationIndex, userInfo, report, round } = params;

  const today = new Date().toLocaleDateString('en-US');
  const bureauName = BUREAU_LEGAL_NAMES[bureau];

  const header = `${userInfo.firstName} ${userInfo.lastName}
${userInfo.addressLine1}
${userInfo.city}, ${userInfo.state} ${userInfo.zip}
Phone: ${userInfo.phone ?? ''}
Email: ${userInfo.email ?? ''}

${today}

${bureauName}
[Insert Current Mailing Address Here]
`;

  const subject = `Re: Request for Investigation of Inaccurate Credit Reporting (Round ${round})
Name: ${userInfo.firstName} ${userInfo.lastName}
SSN (last 4): XXX-XX-${userInfo.ssnLast4 ?? '____'}
DOB: ${userInfo.dob ?? '[MM/DD/YYYY]'}
Credit Report Date: ${report.meta?.reportDate ?? '[Report Date]'}
`;

  const intro = `
To Whom It May Concern:

This letter is a formal dispute and request for reinvestigation under the Fair Credit Reporting Act (FCRA), 15 U.S.C. § 1681i. I am disputing several items on my ${bureauName} credit file that are inaccurate, incomplete, and/or misleading. In many cases, the reporting also appears inconsistent with industry-standard Metro 2® reporting guidelines, which contributes to violations of the FCRA’s "maximum possible accuracy" requirement.

If any furnisher cannot verify the disputed information as fully accurate and properly coded, I request that the tradeline or entry be deleted from my ${bureauName} file.

Below is a list of the specific items I dispute and the reasons for each.
`;

  const issueSections: string[] = [];

  bureauIssues.forEach((issue, index) => {
    const key = makeExplanationKey(issue.ruleId, issue.id);
    const exp = explanationIndex[key] || explanationIndex[issue.ruleId];

    const label = `${index + 1}. ${issue.title}`;

    const accountRef = formatAccountRef(issue.relatedAccount);
    const statutesText = formatStatutes(issue.statutes ?? exp?.statutesCited);

    const coreDescription =
      exp?.letterSnippetBureau ||
      exp?.detailedExplanation ||
      issue.description;

    const block = `
--------------------------------------------------
${label}
--------------------------------------------------
${accountRef}

Dispute Summary:
${issue.description}

Detailed Explanation:
${coreDescription}

Legal Basis:
${statutesText || 'This reporting appears inconsistent with the FCRA’s accuracy requirements and standard Metro 2® reporting practices.'}

Requested Action:
- Conduct a reasonable reinvestigation of this item under 15 U.S.C. § 1681i.
- Correct the reporting so that all balances, past-due amounts, dates, statuses, and remarks are accurate, non-misleading, and consistent across all consumer reporting agencies; OR
- If the information cannot be fully verified with proper coding, delete this entry from my ${bureauName} file.
`;

    issueSections.push(block);
  });

  const outro = `
REQUEST FOR REINVESTIGATION & RESULTS

Pursuant to 15 U.S.C. § 1681i, please conduct a reasonable reinvestigation of each disputed item listed above, including contacting each furnisher of information. For any item that cannot be verified as fully accurate and compliant with industry reporting standards, please delete it from my ${bureauName} credit file.

After completing your reinvestigation, please send me:
1. An updated copy of my ${bureauName} credit report; and
2. A description of the procedures used in your investigation, including the names and addresses of any furnishers you contacted.

I have enclosed copies of my government-issued ID, proof of address, and a copy of the relevant credit report with disputed items highlighted.

This letter is provided for consumer dispute purposes only and is not legal advice.

Sincerely,

_____________________________
${userInfo.firstName} ${userInfo.lastName}
`;

  const plainText = `${header}

${subject}

${intro}

${issueSections.join('\n')}

${outro}
`;

  const html = `<pre>${escapeHtml(plainText)}</pre>`;

  return {
    html,
    plainText,
    recipientName: bureauName,
    recipientType: 'BUREAU',
    bureau
  };
}

${coreDescription}

${userInfo.firstName} ${userInfo.lastName}

// --- Helpers: formatting & escaping ---

function formatAccountRef(ref: { kind: string; creditorName?: string; collectorName?: string; accountId?: string }): string {
  const parts: string[] = [];

  if (ref.kind === 'TRADELINE' && ref.creditorName) {
    parts.push(`Creditor: ${ref.creditorName}`);
  } else if (ref.kind === 'COLLECTION') {
    if (ref.collectorName) parts.push(`Collector: ${ref.collectorName}`);
    if (ref.creditorName) parts.push(`Original Creditor: ${ref.creditorName}`);
  } else if (ref.creditorName || ref.collectorName) {
    parts.push(`Account Party: ${ref.creditorName ?? ref.collectorName}`);
  }

  if (ref.accountId) {
    parts.push(`Internal Account Ref: ${ref.accountId}`);
  }

  if (!parts.length) {
    return 'Account: [Reference details not provided in this summary]';
  }

  return parts.join('\n');
}

function formatStatutes(statutes?: StatuteReference[]): string {
  if (!statutes || !statutes.length) return '';

  const lines = statutes.map((s) => {
    if (s.description) {
      return `${s.law} – ${s.citation}: ${s.description}`;
    }
    return `${s.law} – ${s.citation}`;
  });

  return lines.join('\n');
}

function escapeHtml(text: string): string {
  return text
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;');
}
