🔄 IPN + Upgrade Logic (Updated)
@app.post("/paypal-ipn")
async def paypal_ipn(request: Request):
    form = await request.form()
    email = form.get("custom")
    subscr_id = form.get("subscr_id")
    txn_type = form.get("txn_type")
    amount = float(form.get("payment_gross", "0"))

    if form.get("payment_status") == "Completed":
        user = load_user(email)
        if txn_type == "subscr_signup":  # $99 onboarding
            start = datetime.now()
            expiry = start + relativedelta(months=16)
            user.update({
                "plan": "pro",
                "subscription_id": subscr_id,
                "subscription_start": str(start),
                "expires": str(expiry),
                "subscription_price": amount
            })
            generate_invoice(email, amount)
            send_invoice_email(email)

        elif txn_type == "subscr_payment":  # Recurring $149/month
            generate_invoice(email, amount)
            send_invoice_email(email)

        elif txn_type == "subscr_cancel":
            user["plan"] = "diy"
            user["expires"] = str(datetime.now())
            save_user(email, user)

        save_user(email, user)

    return "OK"
