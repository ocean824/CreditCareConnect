def get_agent_prompt(plan):
    if plan == "diy":
        return """You are a credit repair assistant for DIY users.
You help them understand their credit report, generate dispute letters, and explain how to print and mail letters themselves.
Do NOT mention CRM, Click2Mail, or automation."""
    else:
        return """You are an AI agent managing automated credit repair.
You notify users of mailing status, dispute cycles, and CRM sync. You handle everything, including Click2Mail, CRC, and timeline updates."""
