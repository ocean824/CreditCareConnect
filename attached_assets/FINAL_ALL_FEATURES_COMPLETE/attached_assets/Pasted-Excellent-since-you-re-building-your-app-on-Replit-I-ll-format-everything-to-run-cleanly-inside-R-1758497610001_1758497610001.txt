@app.post("/analyze-report")
async def analyze_credit(file: UploadFile):
    content = await file.read()
    findings = analyze_report(content)
    letters = generate_letters(findings)
    
    # Send via Click2Mail (example: first letter)
    if letters:
        pdf_path = letters[0]["pdf_path"]
        recipient = letters[0]["recipient"]
        send_letter_via_click2mail(pdf_path, recipient)

    return {
        "status": "success",
        "violations": findings,
        "letters": [l["pdf_path"] for l in letters]
    }

5. pdf_utils.py – Jinja2 + WeasyPrint
from jinja2 import Environment, FileSystemLoader
from weasyprint import HTML
import os
from datetime import date

env = Environment(loader=FileSystemLoader("letter_templates"))

def generate_letters(findings):
    letters = []
    for finding in findings:
        template = env.get_template("bureau_dispute.jinja2")
        html = template.render(
            name="John Doe",
            ssn_last4="1234",
            address="123 Main St, FL",
            dob="01/01/1990",
            bureau=finding["bureau"],
            account=finding["account"],
            issue=finding["issue"],
            statutes=", ".join(finding["statutes"]),
            today=str(date.today())
        )
        pdf_path = f"output/{finding['bureau']}_dispute_{finding['account'].replace(' ', '_')}.pdf"
        os.makedirs("output", exist_ok=True)
        HTML(string=html).write_pdf(pdf_path)
        letters.append({"pdf_path": pdf_path, "recipient": {"name": "Equifax", "address1": "PO Box 740241", "city": "Atlanta", "state": "GA", "zip": "30374"}})
    return letters

7. click2mail_api.py
import requests
import base64
import os

def send_letter_via_click2mail(pdf_path, recipient):
    username = os.getenv("CLICK2MAIL_USERNAME")
    password = os.getenv("CLICK2MAIL_PASSWORD")
    auth = base64.b64encode(f"{username}:{password}".encode()).decode()
    
    url = "https://rest.click2mail.com/molpro/1.0/job"
    headers = {
        "Authorization": f"Basic {auth}",
        "Content-Type": "application/x-www-form-urlencoded"
    }

    data = {
        "documentClass": "Letter 8.5 x 11",
        "layout": "Address on First Page",
        "productionTime": "Next Day",
        "envelope": "Double Window",
        "color": "Black and White",
        "paperType": "White 24#",
        "printOption": "Printing One side",
        "mailClass": "First Class",
        "recipients": f"{recipient['name']}\n{recipient['address1']}\n{recipient['city']}, {recipient['state']} {recipient['zip']}"
    }

    with open(pdf_path, "rb") as f:
        files = {
            "file": (os.path.basename(pdf_path), f, "application/pdf")
        }
        response = requests.post(url, headers=headers, data=data, files=files)
        return response.status_code, response.text
