2️⃣ Rules engine: run all rules and never drop a hit

This is the enforcement point. Make analyzeWithRulesEngine:

Iterate all enabled rules.

Run each rule against the whole CreditReport.

Collect all IssueFlags.

Generate AI explanations for every issue (no sampling/skipping).

Return everything; don’t internally filter.

// src/server/services/rulesEngine.ts
import { CreditReport } from '../types/credit';
import { rulesCatalog, RuleConfig } from '../rules/rules_catalog';
import { IssueFlag, AiIssueExplanation } from '../types/issues';
import { runRule_FCRA_INCONSISTENT_BALANCE } from './rules/fcraInconsistentBalance';
import { runRule_METRO2_CHARGEOFF_PAST_DUE_NONZERO } from './rules/metro2ChargeoffPastDue';
import { runRule_FDCPA_COLLECTION_REPORTING_UNVALIDATED } from './rules/fdcpaValidation';
// …import additional rule implementations

export interface AnalyzeWithRulesEngineInput {
  report: CreditReport;
  userId: string;
}

export interface AnalyzeWithRulesEngineOutput {
  issues: IssueFlag[];
  aiExplanations: AiIssueExplanation[];
}

const ruleRunners: Record<string, (report: CreditReport, userId: string) => IssueFlag[]> = {
  FCRA_INCONSISTENT_BALANCE_OR_PAST_DUE: runRule_FCRA_INCONSISTENT_BALANCE,
  METRO2_CHARGEOFF_PAST_DUE_NONZERO: runRule_METRO2_CHARGEOFF_PAST_DUE_NONZERO,
  FDCPA_COLLECTION_REPORTING_UNVALIDATED: runRule_FDCPA_COLLECTION_REPORTING_UNVALIDATED
  // …wire every rule id to its implementation
};

export async function analyzeWithRulesEngine(
  input: AnalyzeWithRulesEngineInput
): Promise<AnalyzeWithRulesEngineOutput> {
  const { report, userId } = input;

  const allIssues: IssueFlag[] = [];

  // 🔒 HARD RULE: iterate EVERY enabled rule, no short-circuiting
  for (const rule of rulesCatalog.rules) {
    if (!rule.enabled) continue;

    const runner = ruleRunners[rule.id];
    if (!runner) {
      // Optional: log that implementation is missing
      continue;
    }

    const issuesFromRule = runner(report, userId) || [];
    // ❗ NEVER drop issues here – push them all
    for (const issue of issuesFromRule) {
      allIssues.push(issue);
    }
  }

  // 🔒 Generate AI explanations for EVERY IssueFlag
  const aiExplanations = await generateAiExplanationsForIssues(allIssues);

  return {
    issues: allIssues,
    aiExplanations
  };
}

3️⃣ Letter generator: include all relevant violations for that target

Next: ensure generateAllDisputeLetters uses every IssueFlag that applies to the selected bureau or collector.

Key rules:

For a bureau letter (Experian, Equifax, TU):

Include all issues whose related tradeline/collection has that bureau in its bureauSnapshots or bureauScope.

For a collector letter:

Include all issues that reference that collector (issue.collectorName or issue.relatedAccount.collectorName).

Do not silently drop low severity issues; you can group them visually, but they must be present.

// src/server/services/disputeLetters.ts
import { CreditReport } from '../types/credit';
import { IssueFlag, AiIssueExplanation } from '../types/issues';

interface GenerateLettersInput {
  report: CreditReport;
  issues: IssueFlag[];
  explanations: AiIssueExplanation[];
  userInfo: {
    name: string;
    address: string;
    phone?: string;
    email?: string;
    ssnLast4?: string;
    dob?: string;
  };
  round: 1 | 2 | 3;
}

export interface DisputeLettersResult {
  bureauLetters: Record<'EXPERIAN' | 'EQUIFAX' | 'TRANSUNION', string | undefined>; // HTML
  collectorLetters: Record<string, string>; // collectorName -> HTML
}

export async function generateAllDisputeLetters(
  input: GenerateLettersInput
): Promise<DisputeLettersResult> {
  const { report, issues, explanations, userInfo, round } = input;

  const bureauLetters: DisputeLettersResult['bureauLetters'] = {
    EXPERIAN: undefined,
    EQUIFAX: undefined,
    TRANSUNION: undefined
  };

  // Helper to match explanations to issues
  const explanationByIssueId = new Map(
    explanations.map((e) => [e.issueId, e])
  );

  // 1) BUREAU LETTERS – include ALL issues relevant to that bureau
  (['EXPERIAN', 'EQUIFAX', 'TRANSUNION'] as const).forEach((bureau) => {
    const bureauIssues = issues.filter((issue) =>
      issue.relatedAccount?.bureauScope?.includes(bureau)
    );

    if (bureauIssues.length === 0) return;

    bureauLetters[bureau] = renderBureauLetterHtml({
      report,
      bureau,
      issues: bureauIssues,
      explanations: bureauIssues.map(
        (issue) => explanationByIssueId.get(issue.id)
      ).filter(Boolean) as AiIssueExplanation[],
      userInfo,
      round
    });
  });

  // 2) COLLECTOR LETTERS – include ALL issues tied to that collector
  const collectorMap = new Map<string, IssueFlag[]>();

  for (const issue of issues) {
    const collectorName =
      issue.collectorName || issue.relatedAccount?.collectorName;
    if (!collectorName) continue;

    if (!collectorMap.has(collectorName)) {
      collectorMap.set(collectorName, []);
    }
    collectorMap.get(collectorName)!.push(issue);
  }

  const collectorLetters: Record<string, string> = {};
  for (const [collectorName, collectorIssues] of collectorMap) {
    const exps = collectorIssues
      .map((issue) => explanationByIssueId.get(issue.id))
      .filter(Boolean) as AiIssueExplanation[];

    collectorLetters[collectorName] = renderCollectorLetterHtml({
      report,
      collectorName,
      issues: collectorIssues,
      explanations: exps,
      userInfo,
      round
    });
  }

  return { bureauLetters, collectorLetters };
}


The key bit: no filtering by severity here. If an issue made it into issues[], it must be considered for the letter. If later you want a “simple mode,” you can add a user option to hide certain categories — but the engine itself shouldn’t silently drop them.
