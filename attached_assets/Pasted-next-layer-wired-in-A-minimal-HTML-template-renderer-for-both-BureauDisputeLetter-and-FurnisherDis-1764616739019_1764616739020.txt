1️⃣ Minimal HTML template for any dispute letter

File: src/server/services/letters/renderLetterHtml.ts

// src/server/services/letters/renderLetterHtml.ts

import { BureauDisputeLetter, FurnisherDisputeLetter } from './generateAllDisputeLetters';

export type AnyLetter = BureauDisputeLetter | FurnisherDisputeLetter;

function escapeHtml(str: string): string {
  return str
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;');
}

/**
 * Very simple HTML wrapper for a dispute letter.
 * - Uses a print-friendly layout
 * - Preserves newlines from the plain-text body
 * - Works for bureau or furnisher letters
 */
export function renderLetterHtml(letter: AnyLetter): string {
  const isBureau = (letter as BureauDisputeLetter).bureau !== undefined;

  const toBlock = letter.toAddressLines
    .map((line) => escapeHtml(line))
    .join('<br>');

  const subjectHtml = escapeHtml(letter.subject);
  const bodyHtml = escapeHtml(letter.body).replace(/\n/g, '<br>');

  const headerLabel = isBureau ? 'Credit Bureau Dispute Letter' : 'Furnisher / Collector Dispute Letter';

  return `<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8" />
  <title>${subjectHtml}</title>
  <style>
    * {
      box-sizing: border-box;
    }
    body {
      font-family: "Times New Roman", Georgia, serif;
      font-size: 12pt;
      line-height: 1.5;
      color: #000;
      margin: 0;
      padding: 0;
    }
    .page {
      width: 8.5in;
      min-height: 11in;
      padding: 1in;
      margin: 0 auto;
    }
    .letter-header-label {
      text-align: right;
      font-size: 10pt;
      color: #555;
      margin-bottom: 0.25in;
    }
    .letter-block {
      margin-bottom: 0.15in;
    }
    .letter-body {
      margin-top: 0.25in;
      white-space: pre-wrap;
    }
  </style>
</head>
<body>
  <div class="page">
    <div class="letter-header-label">${headerLabel}</div>

    <div class="letter-block">
      <!-- The body string already includes sender info and [Date] line -->
      <!-- We just render it as-is -->
      <div class="letter-body">
        ${bodyHtml}
      </div>
    </div>
  </div>
</body>
</html>`;
}


Note: Because your body already includes the consumer header, [Date], subject line text, etc., this template mostly wraps it in a clean, printable layout and preserves line breaks.

2️⃣ Minimal HTML → PDF helper for Replit

File: src/server/services/pdf/htmlToPdf.ts

This uses html-pdf-node, which works nicely in Node/Express.
Install:

npm install html-pdf-node

// src/server/services/pdf/htmlToPdf.ts

// eslint-disable-next-line @typescript-eslint/no-var-requires
const html_to_pdf = require('html-pdf-node');

/**
 * Convert HTML string into a PDF Buffer using html-pdf-node.
 * In Replit, make sure you've installed `html-pdf-node`.
 */
export async function htmlToPdfBuffer(html: string): Promise<Buffer> {
  const file = { content: html };
  const options = {
    format: 'A4',
    printBackground: true,
    margin: {
      top: '10mm',
      right: '10mm',
      bottom: '10mm',
      left: '10mm'
    }
  };

  const pdfBuffer: Buffer = await html_to_pdf.generatePdf(file, options);
  return pdfBuffer;
}


You can later swap this out for Puppeteer, Playwright, or an external PDF API if you prefer—everything else can stay the same.
