Implementing credit‑report parsing and violation detection in Python

Below is an example of a Python module that could be integrated into the backend. It uses pdfplumber to extract text from a PDF credit report and then applies simple heuristics to detect common FCRA violations. The code is simplified for clarity:

import re
import datetime
import pdfplumber
from dataclasses import dataclass, field
from typing import List

@dataclass
class AccountEntry:
    creditor: str
    open_date: str
    balance: float
    status: str
    remarks: str
    original_line: str  # Keep original line for reference

@dataclass
class Violation:
    violation_type: str
    description: str
    accounts: List[AccountEntry] = field(default_factory=list)

def parse_credit_report(pdf_path: str) -> List[AccountEntry]:
    """Extracts rough account information from a credit‑report PDF."""
    accounts: List[AccountEntry] = []
    with pdfplumber.open(pdf_path) as pdf:
        for page in pdf.pages:
            text = page.extract_text() or ""
            # Example pattern matching typical account lines
            # This will vary between bureaus and should be adjusted
            lines = text.split("\n")
            for line in lines:
                if re.search(r"(?i)account\s*#", line):
                    # extract fields using simplistic regexes
                    creditor = re.findall(r"Creditor:\s*(.+?)\s{2,}", line)
                    open_date = re.findall(r"Opened:\s*(\d{2}/\d{2}/\d{4})", line)
                    balance = re.findall(r"Balance:\s*\$?(\d+[\d,]*)", line)
                    status = re.findall(r"Status:\s*([A-Za-z ]+)", line)
                    remarks = re.findall(r"Remarks:\s*(.+)", line)
                    entry = AccountEntry(
                        creditor=creditor[0] if creditor else "Unknown",
                        open_date=open_date[0] if open_date else "",
                        balance=float(balance[0].replace(",", "")) if balance else 0.0,
                        status=status[0] if status else "",
                        remarks=remarks[0] if remarks else "",
                        original_line=line,
                    )
                    accounts.append(entry)
    return accounts

def detect_violations(accounts: List[AccountEntry]) -> List[Violation]:
    """Apply simple FCRA violation rules to a list of accounts."""
    violations: List[Violation] = []
    # 1. Outdated negative information (older than 7 years)
    cutoff_date = datetime.datetime.now() - datetime.timedelta(days=7*365)
    outdated_accounts = []
    for acc in accounts:
        if acc.open_date:
            try:
                opened = datetime.datetime.strptime(acc.open_date, "%m/%d/%Y")
                if opened < cutoff_date and acc.status.lower() in {"delinquent", "charge off"}:
                    outdated_accounts.append(acc)
            except ValueError:
                pass
    if outdated_accounts:
        violations.append(Violation(
            violation_type="Outdated information",
            description=("Negative items must be removed after seven years under the FCRA"),
            accounts=outdated_accounts))
    # 2. Discharged debts misclassified
    misclassified = [acc for acc in accounts if "discharged" in acc.remarks.lower() and acc.status.lower() not in {"discharged", "paid"}]
    if misclassified:
        violations.append(Violation(
            violation_type="Discharged debt misclassification",
            description="Discharged debts should not be reported as active", 
            accounts=misclassified))
    # 3. Duplicate accounts (same creditor and balance)
    seen = {}
    duplicates = []
    for acc in accounts:
        key = (acc.creditor.lower(), acc.balance)
        if key in seen:
            duplicates.append(acc)
        else:
            seen[key] = acc
    if duplicates:
        violations.append(Violation(
            violation_type="Duplicate account entries",
            description="Same debt appears multiple times, which is inaccurate", 
            accounts=duplicates))
    return violations

def generate_dispute_letter(violation: Violation, consumer_name: str, address: str, credit_bureau: str) -> str:
    """Create a dispute letter based on a violation."""
    today = datetime.datetime.now().strftime("%B %d, %Y")
    letter = [
        f"{today}\n",
        f"{credit_bureau}\n",
        "\n",
        f"Re: Request to Investigate and Correct Credit Report Entries\n\n",
        f"Dear Sir or Madam:\n\n",
        f"I am writing to dispute certain information in my credit report. My name is {consumer_name}, and my current mailing address is {address}." 
        " I request an investigation of the following items that I believe are inaccurate or outdated under the Fair Credit Reporting Act (FCRA).\n\n",
    ]
    letter.append(f"Violation type: {violation.violation_type}\n")
    letter.append(f"Explanation: {violation.description}\n\n")
    for acc in violation.accounts:
        letter.append(f"• Creditor: {acc.creditor}, Opened: {acc.open_date}, Balance: ${acc.balance:,.2f}, Status: {acc.status}.\n")
    letter.append("\nI request that you investigate these items and, if you cannot verify their accuracy, remove or correct them within the time required by the FCRA. Please send me a written confirmation of your findings and a free updated copy of my credit report.\n\n")
    letter.append("Thank you for your prompt attention to this matter.\n\n")
    letter.append("Sincerely,\n")
    letter.append(f"{consumer_name}\n")
    letter.append(f"{address}\n")
    return "".join(letter)



The above code can be wired into the backend route that handles uploaded reports. After parsing and detecting violations, iterate over each Violation to generate a dispute letter for the credit bureau and (if needed) a separate letter for the information furnisher (creditor). Strengthening the client–server communication

To ensure the front‑end, analysis service and letter generator work seamlessly:

Use proper HTTP semantics. When an upload or analysis fails, respond with an error status and message rather than returning an empty success response. This helps the front‑end present meaningful error messages.

Validate inputs early. Check that a file was attached, that it is a supported type (PDF or JSON) and that required fields such as consumer name and address are present before parsing. Immediately return a 400 Bad Request if validation fails.

Log exceptions and return problem details. Surround parsing and detection calls with try/except blocks. Log the stack trace to a file or monitoring system, and return a generic message to the client (e.g., “Unable to parse report; please upload a PDF from Experian, Equifax or TransUnion”).

Support multiple report formats. The sample parser above assumes a generic text pattern. In practice, each credit bureau formats their PDF differently. Use separate parsing functions or a configuration file to map field names to each bureau.

CORS and CSRF considerations. If the front‑end is served from a different origin (e.g., app.mydomain.com) than the API (api.mydomain.com), enable CORS on the backend for the front‑end origin. Implement CSRF tokens or SameSite cookies as necessary.
