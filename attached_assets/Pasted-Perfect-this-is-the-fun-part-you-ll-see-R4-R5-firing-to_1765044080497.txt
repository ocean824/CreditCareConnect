3️⃣ From RuleDetection → AiIssueExplanation

Now we wire detections into plain-English explanations that your letter generator can drop into dispute paragraphs.

3.1. Explanation types
// ai.types.ts
import { BureauId, RuleId } from './rules.types';

export type IssueSeverity = 'low' | 'medium' | 'high';

export interface AiIssueExplanation {
  ruleId: RuleId;
  bureau: BureauId;
  accountId: string;
  creditor: string;
  shortTitle: string;        // e.g. "Duplicate LVNV collection on TransUnion"
  summary: string;           // one-sentence human explanation
  lawCitations: string[];    // e.g. ["15 U.S.C. § 1681e(b)"]
  severity: IssueSeverity;
  letterReasonSentence: string; // one sentence to drop directly in a bullet
  letterParagraph: string;      // 1–3 sentences paragraph for letters
}

3.2. Mapper function (switch on ruleId)
// ai.explanations.ts
import { RuleDetection } from './rules.types';
import { AiIssueExplanation } from './ai.types';

export function detectionToAiIssueExplanation(
  det: RuleDetection
): AiIssueExplanation {
  const base = {
    ruleId: det.ruleId,
    bureau: det.bureau,
    accountId: det.accountId,
    creditor: det.creditor
  };

  switch (det.ruleId) {
    case 'R4_METRO2_COLLECTION_REPORTED_AS_OPEN_ACCOUNT': {
      return {
        ...base,
        shortTitle: 'Collector mis-coded as open account',
        summary:
          'This LVNV Funding tradeline is a third-party collection but is coded as an open account with late payments instead of a collection.',
        lawCitations: ['FCRA 15 U.S.C. § 1681e(b)', 'Metro 2 trade vs. collection coding'],
        severity: 'medium',
        letterReasonSentence:
          'LVNV Funding is reporting this debt as an “open” account with late payments even though it is actually a third-party collection, which is misleading and not consistent with industry Metro 2 coding.',
        letterParagraph:
          'LVNV Funding LLC is a third-party debt collector, but on my TransUnion report this tradeline is coded as an open account with 120-day late payments instead of being reported solely as a collection account. That misclassification is misleading and inconsistent with standard Metro 2 reporting guidelines for third-party collections, and it does not meet the FCRA’s requirement of maximum possible accuracy.'
      };
    }

    case 'R5_FCRA_DUPLICATE_COLLECTION_SAME_DEBT': {
      const collectorName = String(det.details.collectorName ?? det.creditor);
      const originalCreditor = String(
        det.details.originalCreditor ?? 'the original creditor'
      );
      const balance = det.details.balance as number | undefined;
      const duplicateIds = (det.details.duplicateTradelines as string[]) || [];

      const balanceText =
        typeof balance === 'number' ? ` for approximately $${balance}` : '';

      return {
        ...base,
        shortTitle: 'Duplicate collection for same debt',
        summary:
          'This debt is being reported more than once on the same bureau by the same collector for the same original creditor and balance.',
        lawCitations: ['FCRA 15 U.S.C. § 1681e(b)'],
        severity: 'high',
        letterReasonSentence:
          `${collectorName} is reporting the same ${originalCreditor} collection account more than once on the same credit report, which unfairly doubles the negative impact of a single alleged debt.`,
        letterParagraph:
          `${collectorName} is reporting the same ${originalCreditor} collection account${balanceText} more than once on my ${det.bureau} file. These duplicate tradelines appear to represent a single alleged obligation but show up as separate derogatory accounts (IDs: ${duplicateIds.join(
            ', '
          )}). Reporting the same debt multiple times is misleading and does not satisfy the FCRA requirement that consumer reports be prepared with maximum possible accuracy. One of these entries should be removed so that only a single, accurate tradeline remains, if any.`
      };
    }

    // examples for earlier rules so your system is complete
    case 'R2_FCRA_BALANCE_INCONSISTENT_ACROSS_BUREAUS': {
      return {
        ...base,
        shortTitle: 'Inconsistent balance across bureaus',
        summary:
          'This account shows different balances and/or past-due amounts across the three credit bureaus.',
        lawCitations: ['FCRA 15 U.S.C. § 1681e(b)'],
        severity: 'high',
        letterReasonSentence:
          'The reported balance and past-due amounts for this account are inconsistent across the three credit bureaus, which cannot all be simultaneously accurate.',
        letterParagraph:
          'For this account, the balance and/or past-due amounts differ materially between my Experian, Equifax, and TransUnion reports. The same loan cannot simultaneously show different balances and past-due amounts across bureaus. This inconsistency makes the reporting misleading and fails to meet the FCRA’s “maximum possible accuracy” standard. The tradeline must be corrected so that all bureaus reflect the same, verifiably accurate figures, or it should be removed.'
      };
    }

    default:
      // Fallback – still safe for unknown future rules
      return {
        ...base,
        shortTitle: 'Credit reporting issue detected',
        summary:
          'An automated rule detected a potential accuracy or Metro 2 coding problem with this tradeline.',
        lawCitations: [],
        severity: 'low',
        letterReasonSentence:
          'There is a potential inaccuracy or coding error in the way this account is being reported.',
        letterParagraph:
          'My automated review of this account identified a potential inaccuracy or Metro 2 coding issue in the way this tradeline is being reported. I request that you conduct a reasonable reinvestigation, verify all data fields and coding for this account, and correct or delete any information that cannot be verified with complete accuracy.'
      };
  }
}

4️⃣ Group explanations → dispute-letter paragraphs

Now we combine all detections for a given bureau + account into a single block of text you can drop into a letter.

4.1. Helper type for grouping
// ai.grouping.ts
import { RuleDetection } from './rules.types';
import { AiIssueExplanation } from './ai.types';
import { detectionToAiIssueExplanation } from './ai.explanations';
import { BureauId } from './rules.types';

export interface AccountLetterSection {
  bureau: BureauId;
  accountId: string;
  creditor: string;
  shortHeader: string;
  reasons: string[];   // bullet sentences
  paragraphs: string[]; // multi-sentence paragraphs
}

4.2. Build sections per account (per bureau)
export function buildLetterSectionsForBureau(
  bureau: BureauId,
  detections: RuleDetection[]
): AccountLetterSection[] {
  // filter by bureau
  const byBureau = detections.filter((d) => d.bureau === bureau);

  // group by creditor + accountId (or accountGroupId if you add it to details)
  const groups = new Map<string, RuleDetection[]>();

  for (const det of byBureau) {
    const key = `${det.creditor}::${det.accountId}`;
    if (!groups.has(key)) groups.set(key, []);
    groups.get(key)!.push(det);
  }

  const sections: AccountLetterSection[] = [];

  for (const [key, group] of groups.entries()) {
    const [creditor] = key.split('::');
    const explanations: AiIssueExplanation[] = group.map((d) =>
      detectionToAiIssueExplanation(d)
    );

    // de-dupe by ruleId so you don't repeat the same rule paragraph twice
    const uniqueByRule = new Map<string, AiIssueExplanation>();
    for (const exp of explanations) {
      if (!uniqueByRule.has(exp.ruleId)) {
        uniqueByRule.set(exp.ruleId, exp);
      }
    }

    const uniqueExps = Array.from(uniqueByRule.values());

    const shortHeader =
      uniqueExps.length === 1
        ? uniqueExps[0].shortTitle
        : `${creditor} – multiple reporting issues detected`;

    sections.push({
      bureau,
      accountId: group[0].accountId,
      creditor,
      shortHeader,
      reasons: uniqueExps.map((e) => e.letterReasonSentence),
      paragraphs: uniqueExps.map((e) => e.letterParagraph)
    });
  }

  return sections;
}
