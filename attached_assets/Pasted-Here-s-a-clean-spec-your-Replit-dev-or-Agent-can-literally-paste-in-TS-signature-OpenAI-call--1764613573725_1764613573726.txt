// server/services/creditReportParser.ts

export function buildIdentityIqUserPrompt(params: {
  rawText: string;
  bureauHint: BureauCode;
  provider?: string;
  userId: string;
  existingReportId?: string;
}): string {
  const { rawText, bureauHint, provider = 'IDENTITYIQ', userId, existingReportId } = params;

  // To avoid token blowup, you can slice or chunk rawText,
  // but here we simply slice to a large limit. Replit can refine later.
  const snippet = rawText.length > 16000 ? rawText.slice(0, 16000) : rawText;

  return `
You will be given the raw text of a consumer credit report.

Provider hint: ${provider}
Bureau hint: ${bureauHint}
User ID: ${userId}
Existing report ID (optional): ${existingReportId ?? 'none'}

If bureauHint == "TRI":
- Assume this is a tri-merge report that includes Experian, Equifax, and TransUnion data.
- IdentityIQ often labels sections clearly (e.g., "Experian", "Equifax", "TransUnion" headers).
- You MUST:
  - Identify the three bureau sections,
  - Match tradelines and collections that refer to the same underlying account across bureaus,
  - Populate tradelines[].bureaus and tradelines[].bureauSnapshots accordingly,
  - Populate collections[].bureaus and collections[].bureauSnapshots accordingly,
  - Set meta.bureausIncluded to the list of bureaus actually present (usually all three).

If bureauHint is a single bureau (EXPERIAN/EQUIFAX/TRANSUNION):
- Assume this text only reflects that bureau.
- meta.bureausIncluded should be [that bureau].
- All tradelines/collections/inquiries should have bureaus / bureauSnapshots limited to that one bureau.

When you construct the JSON:

- Set creditReport.id to existingReportId if provided; otherwise use a simple string (e.g., "report-1").
- Set creditReport.userId to the userId provided above.
- meta.provider should be "${provider}" unless the raw text clearly shows a different provider.
- meta.reportDate should come from the report date in the text (IdentityIQ usually shows it on the first pages). Use "YYYY-MM-DD". If you cannot determine a date, use "".

RAW REPORT TEXT (POSSIBLY TRUNCATED FOR LENGTH):

"""${snippet}"""

Now return ONLY a single JSON object matching the CreditReport structure described in the system prompt. Do NOT include any explanations or commentary.
`;
}

// OpenAI client
const openai = new OpenAI({
  apiKey: process.env.OPENAI_API_KEY!
});

export async function parseCreditReportWithAiOnly(
  params: ParseCreditReportWithAiOnlyParams
): Promise<CreditReport> {
  const { rawText, bureauHint, provider = 'IDENTITYIQ', userId, existingReportId } = params;

  const systemPrompt = IDENTITYIQ_TRI_MERGE_SYSTEM_PROMPT;
  const userPrompt = buildIdentityIqUserPrompt({
    rawText,
    bureauHint,
    provider,
    userId,
    existingReportId
  });

  const response = await openai.responses.create({
    model: process.env.OPENAI_MODEL || 'gpt-5.1-chat',
    input: [
      { role: 'system', content: systemPrompt },
      { role: 'user', content: userPrompt }
    ],
    response_format: {
      type: 'json_schema',
      json_schema: CREDIT_REPORT_JSON_SCHEMA
    }
  });

  const first = response.output[0].content[0];
  if (first.type !== 'output_json') {
    throw new Error('Unexpected OpenAI response format: expected output_json');
  }

  const parsed = first.parsed as unknown as CreditReport;

  // 🔒 Minimal sanity checks; Replit can add zod validation here
  if (!parsed || !parsed.meta || !parsed.tradelines || !Array.isArray(parsed.tradelines)) {
    throw new Error('Invalid CreditReport JSON from AI parser');
  }

  // Optionally override ids with your own UUIDs here if you want
  // before saving to DB.

  return parsed;
}
