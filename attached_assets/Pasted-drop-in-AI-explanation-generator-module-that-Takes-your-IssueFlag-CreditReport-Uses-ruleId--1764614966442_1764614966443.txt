drop-in AI explanation generator module that:

Takes your IssueFlag[] + CreditReport.

Uses ruleId, category, lawReferences, description, and details.

Calls OpenAI to return letter-ready, FCRA/Metro 2/FDCPA-flavored explanations for each issue.

You can tweak paths/model names, but this is ready for Replit.

// src/server/services/ai/aiIssueExplanation.ts

import OpenAI from 'openai';
import { CreditReport, Tradeline, CollectionAccount, PublicRecord } from '../types/credit';
import { IssueFlag, AiIssueExplanation } from '../rules/ruleImplementations'; // adjust path

const openai = new OpenAI({
  apiKey: process.env.OPENAI_API_KEY
});

// You can override this via env if you want:
const ISSUE_MODEL = process.env.OPENAI_ISSUE_MODEL ?? 'gpt-4.1-mini';

export interface GenerateAiExplanationsParams {
  report: CreditReport;
  issues: IssueFlag[];
  userId: string;
}

/**
 * Top-level function your rules engine calls after collecting IssueFlag[].
 * Returns one AiIssueExplanation per issueId.
 */
export async function generateAiExplanationsForIssues(
  params: GenerateAiExplanationsParams
): Promise<AiIssueExplanation[]> {
  const { report, issues } = params;

  const explanations: AiIssueExplanation[] = [];

  for (const issue of issues) {
    const context = buildIssueContext(issue, report);

    const { summary, fullText, raw } = await callOpenAiForIssueExplanation(issue, context);

    explanations.push({
      issueId: issue.id,
      model: ISSUE_MODEL,
      summary,
      fullText,
      rawJson: raw
    });
  }

  return explanations;
}

// ---------------------------------------------------------------------
// Context building helpers
// ---------------------------------------------------------------------

function findTradeline(report: CreditReport, id?: string): Tradeline | undefined {
  if (!id) return undefined;
  return report.tradelines.find((t) => t.id === id);
}

function findCollection(report: CreditReport, id?: string): CollectionAccount | undefined {
  if (!id) return undefined;
  return report.collections.find((c) => c.id === id);
}

function findPublicRecord(report: CreditReport, id?: string): PublicRecord | undefined {
  if (!id) return undefined;
  return report.publicRecords.find((p) => p.id === id);
}

/**
 * Build a compact, safe context blob to send to the model.
 * We do NOT send the entire report, just the fields needed for a good explanation.
 */
function buildIssueContext(issue: IssueFlag, report: CreditReport) {
  const ctx: any = {
    ruleId: issue.ruleId,
    category: issue.category,
    lawReferences: issue.lawReferences,
    issueDescription: issue.description,
    issueDetails: issue.details ?? {},
    relatedAccount: issue.relatedAccount,
    reportMeta: {
      reportDate: report.meta?.reportDate ?? null,
      pulledFrom: report.meta?.pulledFrom ?? null
    }
  };

  if (issue.relatedAccount.type === 'TRADELINE') {
    const tl = findTradeline(report, issue.relatedAccount.tradelineId);
    if (tl) {
      ctx.tradeline = {
        id: tl.id,
        creditorName: tl.creditorName,
        type: tl.type,
        status: tl.status,
        balance: tl.balance,
        pastDue: tl.pastDue,
        bureaus: tl.bureaus,
        remarks: tl.remarks,
        bureauSnapshots: tl.bureauSnapshots?.map((b) => ({
          bureau: b.bureau,
          status: b.status,
          balance: b.balance,
          pastDue: b.pastDue,
          remarks: b.remarks
        }))
      };
    }
  }

  if (issue.relatedAccount.type === 'COLLECTION') {
    const col = findCollection(report, issue.relatedAccount.collectionId);
    if (col) {
      ctx.collection = {
        id: col.id,
        collectorName: col.collectorName,
        originalCreditorName: col.originalCreditorName,
        status: col.status,
        balance: col.balance,
        bureaus: col.bureaus,
        remarks: col.remarks,
        bureauSnapshots: col.bureauSnapshots?.map((b) => ({
          bureau: b.bureau,
          status: b.status,
          balance: b.balance,
          pastDue: b.pastDue,
          remarks: b.remarks
        }))
      };
    }
  }

  if (issue.relatedAccount.type === 'PUBLIC_RECORD') {
    const pr = findPublicRecord(report, issue.relatedAccount.publicRecordId);
    if (pr) {
      ctx.publicRecord = {
        id: pr.id,
        type: pr.type,
        amount: pr.amount,
        courtName: pr.courtName,
        filedDate: pr.filedDate,
        bureaus: pr.bureaus,
        remarks: pr.remarks
      };
    }
  }

  return ctx;
}

// ---------------------------------------------------------------------
// OpenAI call + prompt
// ---------------------------------------------------------------------

interface ModelIssueExplanationResponse {
  summary: string;
  fullText: string;
}

/**
 * Calls OpenAI once per issue, asking for JSON:
 * {
 *   "summary": "...",
 *   "fullText": "..."
 * }
 */
async function callOpenAiForIssueExplanation(
  issue: IssueFlag,
  context: any
): Promise<{ summary: string; fullText: string; raw: unknown }> {
  // Hard fallback in case OpenAI is not configured
  if (!process.env.OPENAI_API_KEY) {
    const fallbackText =
      'This account appears to have a potential reporting issue under consumer protection laws. ' +
      'Please review the attached details carefully and consider disputing any inaccurate information.';
    return {
      summary: 'Possible reporting violation identified for this account.',
      fullText: fallbackText,
      raw: { reason: 'NO_API_KEY' }
    };
  }

  const completion = await openai.chat.completions.create({
    model: ISSUE_MODEL,
    temperature: 0.25,
    response_format: { type: 'json_object' },
    messages: [
      {
        role: 'system',
        content: [
          'You are an expert credit-report compliance analyst.',
          'Your job is to explain, in plain consumer-friendly language, why a specific item on a credit report may violate FCRA, FDCPA, and/or Metro 2 guidelines.',
          'You MUST:',
          '- Use the ruleId, category, lawReferences, and the issue details provided.',
          '- Be precise about what looks wrong and how it could harm the consumer.',
          '- Refer to laws by name/section (e.g., FCRA 15 U.S.C. § 1681e(b), FDCPA 15 U.S.C. § 1692e) when they are given.',
          '- Write in neutral, explanatory tone. No legal advice. No instructions like “you should sue.”',
          '- Write as if the text will be pasted directly inside a dispute letter paragraph.',
          '- Do NOT include greetings, closings, or ask questions.',
          '- Do NOT say “in this letter” or “I” or “we.” Write about “this account” or “this reporting.”'
        ].join('\n')
      },
      {
        role: 'user',
        content: buildIssuePromptUserContent(issue, context)
      }
    ]
  });

  const rawContent = completion.choices[0]?.message?.content ?? '{}';

  let parsed: ModelIssueExplanationResponse;
  try {
    parsed = JSON.parse(rawContent) as ModelIssueExplanationResponse;
  } catch {
    // Fallback if the model ever misbehaves
    parsed = {
      summary: 'Possible FCRA/FDCPA/Metro 2 issue with this account’s reporting.',
      fullText:
        'There appears to be a mismatch or inconsistency in how this account is reported compared with legal and industry standards. ' +
        'The consumer may wish to dispute the accuracy and completeness of this tradeline or collection account with the credit bureaus and the furnisher.'
    };
  }

  const summary = (parsed.summary || '').trim();
  const fullText = (parsed.fullText || '').trim();

  return { summary, fullText, raw: safelyAttachRaw(rawContent) };
}

function buildIssuePromptUserContent(issue: IssueFlag, context: any): string {
  return JSON.stringify(
    {
      instructions: {
        goal:
          'Explain this single suspected violation in 1–2 sentences of summary and 2–4 sentences of full explanation, ready for a dispute letter.',
        style: {
          tone: 'formal but plain language',
          perspective: 'third-person about the consumer',
          noLegalAdvice: true,
          noGreetingOrClosing: true,
          mentionLawsByName: true
        },
        outputFormat: {
          summary:
            '1–2 sentence, consumer-friendly description of what is wrong with this account’s reporting.',
          fullText:
            '2–4 sentence explanation that expands on the summary, mentions specific laws/sections provided, and clearly states why the reporting is inaccurate, incomplete, inconsistent, or misleading.'
        }
      },
      issue: {
        id: issue.id,
        ruleId: issue.ruleId,
        category: issue.category,
        lawReferences: issue.lawReferences,
        severity: issue.severity,
        description: issue.description,
        details: issue.details
      },
      relatedAccount: context.relatedAccount,
      reportMeta: context.reportMeta,
      // tradeline/collection/publicRecord are optional; include if present
      tradeline: context.tradeline ?? null,
      collection: context.collection ?? null,
      publicRecord: context.publicRecord ?? null
    },
    null,
    2
  );
}

function safelyAttachRaw(content: string): unknown {
  try {
    return JSON.parse(content);
  } catch {
    return { rawText: content };
  }
}

How this fits into your pipeline

Rules engine runs all 24 rule implementations → returns IssueFlag[].

Call:

const { issues } = await analyzeWithRulesEngine({ report, userId });
const aiExplanations = await generateAiExplanationsForIssues({ report, issues, userId });


Store issues in violations table and aiExplanations in violation_explanations.

Your generateAllDisputeLetters function pulls:

IssueFlag (for ruleId, category, law refs, etc.),

AiIssueExplanation.fullText (paragraphs),

Groups them per account/bureau/collector,

And drops the explanation paragraphs straight into the body of each letter.
