This version handles 4 high-impact rules:

METRO2_CHARGEOFF_PASTDUE_NONZERO

METRO2_SOLD_ACCOUNT_STILL_SHOWING_BALANCE

METRO2_DUPLICATE_TRADELINES_SAME_DEBT

// server/services/advancedCreditAnalysis.ts

import crypto from 'crypto';
import OpenAI from 'openai';

import rulesCatalog from '../config/rules_catalog.json';
import {
  CreditReport,
  IssueFlag,
  AiIssueExplanation,
  IssueSeverity,
  IssueCategory,
  RelatedAccountRef,
  StatuteReference
} from '../types/credit';

const apiKey = process.env.OPENAI_API_KEY;
const openai = apiKey ? new OpenAI({ apiKey }) : null;

interface AnalyzeInput {
  report: CreditReport;
  userId: string;
}

interface AnalyzeOutput {
  issues: IssueFlag[];
  aiExplanations: AiIssueExplanation[];
}

type RuleConfig = (typeof rulesCatalog)['rules'][number];

const ruleIndex = new Map<string, RuleConfig>();
rulesCatalog.rules.forEach((r) => ruleIndex.set(r.id, r));

export async function analyzeWithRulesEngine(
  input: AnalyzeInput
): Promise<AnalyzeOutput> {
  const { report, userId } = input;
  const issues: IssueFlag[] = [];

  // 1️⃣ METRO2_CHARGEOFF_PASTDUE_NONZERO
  issues.push(...detectChargeoffPastDueNonZero(report, userId));

  // 2️⃣ METRO2_SOLD_ACCOUNT_STILL_SHOWING_BALANCE
  issues.push(...detectSoldStillShowingBalance(report, userId));

  // 3️⃣ METRO2_DUPLICATE_TRADELINES_SAME_DEBT
  issues.push(...detectDuplicateTradelinesSameDebt(report, userId));

  // 4️⃣ FDCPA_DUPLICATE_COLLECTORS_SAME_DEBT
  issues.push(...detectDuplicateCollectorsSameDebt(report, userId));

  // 5️⃣ Ask OpenAI to explain issues (if configured)
  const aiExplanations = openai
    ? await generateAiExplanationsForIssues({ issues, report, userId })
    : [];

  return { issues, aiExplanations };
}

/* ------------------------------------------------------------------ */
/* Rule 1: METRO2_CHARGEOFF_PASTDUE_NONZERO                           */
/* ------------------------------------------------------------------ */

function detectChargeoffPastDueNonZero(
  report: CreditReport,
  userId: string
): IssueFlag[] {
  const ruleId = 'METRO2_CHARGEOFF_PASTDUE_NONZERO';
  const cfg = ruleIndex.get(ruleId);
  if (!cfg) return [];

  const found: IssueFlag[] = [];

  for (const tl of report.tradelines) {
    if (tl.status !== 'CHARGE_OFF') continue;
    if (!tl.pastDue || tl.pastDue <= 0) continue;

    const id = crypto.randomUUID();
    const related: RelatedAccountRef = {
      kind: 'TRADELINE',
      accountId: tl.id,
      creditorName: tl.creditorName,
      bureauScope: tl.bureaus
    };

    const statutes: StatuteReference[] = (cfg.law || []).map((l) => ({
      law: l.law as StatuteReference['law'],
      citation: l.citation,
      description: l.description
    }));

    const issue: IssueFlag = {
      id,
      userId,
      reportId: report.id,
      ruleId,
      category: cfg.category as IssueCategory,
      severity: cfg.severityDefault as IssueSeverity,
      title: cfg.label,
      description:
        cfg.description +
        ` Example: ${tl.creditorName} is coded as charge-off with past-due = ${tl.pastDue} and balance = ${tl.balance}.`,
      relatedAccount: related,
      statutes,
      disputeRecommended: true,
      isCollectorIssue: false,
      collectorName: undefined,
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString()
    };

    found.push(issue);
  }

  return found;
}

/* Rule 2: METRO2_SOLD_ACCOUNT_STILL_SHOWING_BALANCE                  */
/* ------------------------------------------------------------------ */

function detectSoldStillShowingBalance(
  report: CreditReport,
  userId: string
): IssueFlag[] {
  const ruleId = 'METRO2_SOLD_ACCOUNT_STILL_SHOWING_BALANCE';
  const cfg = ruleIndex.get(ruleId);
  if (!cfg) return [];

  const found: IssueFlag[] = [];

  for (const tl of report.tradelines) {
    const remarksText = (tl.remarks || []).join(' ').toLowerCase();
    const sold = remarksText.includes('sold to') || remarksText.includes('purchased by another lender');

    if (!sold) continue;
    if (!tl.balance || tl.balance <= 0) continue; // sold accounts should typically show 0 balance

    const id = crypto.randomUUID();
    const related: RelatedAccountRef = {
      kind: 'TRADELINE',
      accountId: tl.id,
      creditorName: tl.creditorName,
      bureauScope: tl.bureaus
    };

    const statutes: StatuteReference[] = (cfg.law || []).map((l) => ({
      law: l.law as StatuteReference['law'],
      citation: l.citation,
      description: l.description
    }));

    const issue: IssueFlag = {
      id,
      userId,
      reportId: report.id,
      ruleId,
      category: cfg.category as IssueCategory,
      severity: cfg.severityDefault as IssueSeverity,
      title: cfg.label,
      description:
        cfg.description +
        ` Example: ${tl.creditorName} shows 'sold/purchased' in remarks but still reports a non-zero balance of ${tl.balance}.`,
      relatedAccount: related,
      statutes,
      disputeRecommended: true,
      isCollectorIssue: false,
      collectorName: undefined,
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString()
    };

    found.push(issue);
  }

  return found;
}

/* ------------------------------------------------------------------ */
/* Rule 3: METRO2_DUPLICATE_TRADELINES_SAME_DEBT                      */
/* ------------------------------------------------------------------ */

function detectDuplicateTradelinesSameDebt(
  report: CreditReport,
  userId: string
): IssueFlag[] {
  const ruleId = 'METRO2_DUPLICATE_TRADELINES_SAME_DEBT';
  const cfg = ruleIndex.get(ruleId);
  if (!cfg) return [];

  const found: IssueFlag[] = [];

  // Very simple grouping heuristic:
  // key = originalCreditorName (or creditorName) + rounded balance
  const groups = new Map<string, { key: string; items: { id: string; name: string; balance: number | undefined }[] }>();

  for (const col of report.collections) {
    const baseName = (col.originalCreditorName || col.collectorName || '').toLowerCase().trim();
    const balanceRounded = col.balance != null ? Math.round(col.balance) : 0;
    const key = `${baseName}|${balanceRounded}`;

    if (!groups.has(key)) {
      groups.set(key, { key, items: [] });
    }
    groups.get(key)!.items.push({
      id: col.id,
      name: col.collectorName || '',
      balance: col.balance
    });
  }

  for (const [, group] of groups) {
    if (group.items.length <= 1) continue; // not a duplicate

    // Flag once per group
    const id = crypto.randomUUID();
    const first = group.items[0];

    const related: RelatedAccountRef = {
      kind: 'COLLECTION',
      accountId: first.id,
      collectorName: first.name
    };

    const statutes: StatuteReference[] = (cfg.law || []).map((l) => ({
      law: l.law as StatuteReference['law'],
      citation: l.citation,
      description: l.description
    }));

    const creditors = Array.from(
      new Set(group.items.map((i) => i.name || 'Unknown Collector'))
    ).join(', ');

    const desc =
      cfg.description +
      ` This report appears to show ${group.items.length} collection tradelines for the same underlying debt (same original creditor/balance) across collectors: ${creditors}.`;

    const issue: IssueFlag = {
      id,
      userId,
      reportId: report.id,
      ruleId,
      category: cfg.category as IssueCategory,
      severity: cfg.severityDefault as IssueSeverity,
      title: cfg.label,
      description: desc,
      relatedAccount: related,
      statutes,
      disputeRecommended: true,
      isCollectorIssue: true,
      collectorName: creditors,
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString()
    };

    found.push(issue);
  }

  return found;
}

/* ------------------------------------------------------------------ */
/* Rule 4: FDCPA_DUPLICATE_COLLECTORS_SAME_DEBT                       */
/* ------------------------------------------------------------------ */

function detectDuplicateCollectorsSameDebt(
  report: CreditReport,
  userId: string
): IssueFlag[] {
  const ruleId = 'FDCPA_DUPLICATE_COLLECTORS_SAME_DEBT';
  const cfg = ruleIndex.get(ruleId);
  if (!cfg) return [];

  const found: IssueFlag[] = [];

  // Group by original creditor + rounded balance
  const groups = new Map<string, { key: string; collectors: string[]; entries: typeof report.collections }>();

  for (const col of report.collections) {
    const baseName = (col.originalCreditorName || '').toLowerCase().trim();
    const balanceRounded = col.balance != null ? Math.round(col.balance) : 0;
    const key = `${baseName}|${balanceRounded}`;

    if (!groups.has(key)) {
      groups.set(key, { key, collectors: [], entries: [] as any });
    }
    const group = groups.get(key)!;
    group.entries.push(col);
    if (col.collectorName) {
      group.collectors.push(col.collectorName);
    }
  }

  for (const [, group] of groups) {
    const uniqueCollectors = Array.from(
      new Set(group.collectors.map((name) => name.toLowerCase().trim()))
    );
    if (uniqueCollectors.length <= 1) continue; // only 1 collector → not a problem for this rule

    // Flag once per group
    const id = crypto.randomUUID();
    const first = group.entries[0];

    const related: RelatedAccountRef = {
      kind: 'COLLECTION',
      accountId: first.id,
      collectorName: first.collectorName,
      bureauScope: first.bureaus
    };

    const statutes: StatuteReference[] = (cfg.law || []).map((l) => ({
      law: l.law as StatuteReference['law'],
      citation: l.citation,
      description: l.description
    }));

    const collectorList = Array.from(
      new Set(group.collectors)
    ).join(', ');

    const desc =
      cfg.description +
      ` Example: the same underlying debt (original creditor ${first.originalCreditorName ?? 'unknown'}) with balance ~${first.balance} appears to be handled by multiple collectors at once: ${collectorList}.`;

    const issue: IssueFlag = {
      id,
      userId,
      reportId: report.id,
      ruleId,
      category: cfg.category as IssueCategory,
      severity: cfg.severityDefault as IssueSeverity,
      title: cfg.label,
      description: desc,
      relatedAccount: related,
      statutes,
      disputeRecommended: true,
      isCollectorIssue: true,
      collectorName: collectorList,
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString()
    };

    found.push(issue);
  }

  return found;
}

/* AI Explanations (OpenAI)                                           */
/* ------------------------------------------------------------------ */

async function generateAiExplanationsForIssues(params: {
  issues: IssueFlag[];
  report: CreditReport;
  userId: string;
}): Promise<AiIssueExplanation[]> {
  const { issues, report, userId } = params;
  if (!openai) return [];

  const explanations: AiIssueExplanation[] = [];

  for (const issue of issues) {
    const ruleCfg = ruleIndex.get(issue.ruleId);
    const statutes = ruleCfg?.law ?? [];

    const systemPrompt = `
You are an expert consumer credit and FDCPA/FCRA compliance analyst.
You receive (1) a machine-detected issue and (2) some context.
Return a short, precise JSON object explaining the issue in plain English for use in dispute letters.
Do NOT include legal advice or litigation strategy, just explanation.
`;

    const userPrompt = `
ISSUE:
${JSON.stringify(issue, null, 2)}

REPORT META:
${JSON.stringify(report.meta ?? {}, null, 2)}

RULE DESCRIPTION:
${ruleCfg?.description ?? ''}

STATUTES:
${JSON.stringify(statutes, null, 2)}

RESPONSE FORMAT (JSON ONLY):
{
  "shortSummary": "1–2 sentence plain-English summary.",
  "detailedExplanation": "A paragraph explaining what appears wrong with this reporting, focusing on FCRA/Metro 2/FDCPA concepts.",
  "disputeTheory": "One-sentence summary of why this should be disputed.",
  "statutesCited": [
    { "law": "FCRA" | "FDCPA" | "METRO2" | "OTHER", "citation": "string", "description": "string" }
  ],
  "letterSnippetBureau": "Optional text chunk tailored to be used in a bureau dispute letter.",
  "letterSnippetCollector": "Optional text chunk tailored to be used in a collector/FDCPA letter.",
  "recommendedEvidence": "Optional suggestion of documents the consumer could attach."
}
`;

    try {
      const response = await openai.responses.create({
        model: process.env.OPENAI_MODEL || 'gpt-5.1-chat-latest',
        input: [
          { role: 'system', content: systemPrompt },
          { role: 'user', content: userPrompt }
        ],
        response_format: { type: 'json_object' }
      });

      const content = response.output[0].content[0];
      if (content.type !== 'output_text') {
        throw new Error('Unexpected OpenAI response format');
      }

      const parsed = JSON.parse(content.text());

      const aiExp: AiIssueExplanation = {
        id: crypto.randomUUID(),
        userId,
        reportId: report.id,
        ruleId: issue.ruleId,
        issueId: issue.id,
        shortSummary: parsed.shortSummary || issue.title,
        detailedExplanation: parsed.detailedExplanation || issue.description,
        disputeTheory: parsed.disputeTheory,
        statutesCited: (parsed.statutesCited || statutes).map((s: any) => ({
          law: s.law,
          citation: s.citation,
          description: s.description
        })),
        letterSnippetBureau: parsed.letterSnippetBureau,
        letterSnippetCollector: parsed.letterSnippetCollector,
        recommendedEvidence: parsed.recommendedEvidence,
        model: process.env.OPENAI_MODEL || 'gpt-5.1-chat-latest',
        createdAt: new Date().toISOString()
      };

      explanations.push(aiExp);
    } catch (err) {
      console.error('Error generating AI explanation for issue', issue.id, err);

      // Fallback explanation if AI fails
      const fallback: AiIssueExplanation = {
        id: crypto.randomUUID(),
        userId,
        reportId: report.id,
        ruleId: issue.ruleId,
        issueId: issue.id,
        shortSummary: issue.title,
        detailedExplanation: issue.description,
        disputeTheory: ruleCfg?.description,
        statutesCited: (ruleCfg?.law || []).map((l) => ({
          law: l.law as StatuteReference['law'],
          citation: l.citation,
          description: l.description
        })),
        model: 'fallback',
        createdAt: new Date().toISOString()
      };

      explanations.push(fallback);
    }
  }

  return explanations;
}
