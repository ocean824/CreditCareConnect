2️⃣ Patch /api/credit-reports/upload

Goal: when a user uploads a report:

Read & OCR/parse text

Use AI-only parser (parseCreditReportWithAiOnly)

Run rules engine + AI explanations

Persist report + issues + explanations

Here’s a sketch of the route handler:

// routes/creditReports.ts (or wherever your routes live)
import { Router, Request, Response } from 'express';
import { parseCreditReportWithAiOnly } from '../services/creditReportParser';
import { analyzeWithRulesEngine } from '../services/advancedCreditAnalysis';
import { ocrPdfOrImage } from '../services/ocr';
import { requireAuth } from '../middleware/auth';
// import db from '../db'; // whatever your ORM instance is

const router = Router();

/**
 * POST /api/credit-reports/upload
 * Body:
 *   filePath or fileUrl  (depending on how you store uploads)
 *   bureau: 'EXPERIAN' | 'EQUIFAX' | 'TRANSUNION' | 'TRI'
 */
router.post(
  '/upload',
  requireAuth,
  async (req: Request, res: Response) => {
    try {
      const userId = req.user!.id; // adapt to your auth type
      const { filePath, bureau = 'TRI' } = req.body;

      if (!filePath) {
        return res.status(400).json({ error: 'filePath is required.' });
      }

      // 1️⃣ OCR or direct text extraction
      const rawText = await ocrPdfOrImage(filePath);

      // 2️⃣ Force advanced AI parser – throws if no OpenAI key
      const parsedReport = await parseCreditReportWithAiOnly(rawText, bureau);

      // 3️⃣ Run rules + AI explanations
      const analysisResult = await analyzeWithRulesEngine({
        report: parsedReport,
        userId
      });
      // assume this returns:
      // { issues: IssueFlag[], aiExplanations: AiIssueExplanation[] }

      // 4️⃣ Persist to DB – adapt to your ORM
      // 4a. Save the credit report
      const savedReport = await db.creditReports.create({
        data: {
          userId,
          bureau,
          sourceFilePath: filePath,
          rawText,
          parsedJson: parsedReport,
          analyzedAt: new Date()
        }
      });

      // 4b. Save issues (violations)
      for (const issue of analysisResult.issues) {
        await db.violations.create({
          data: {
            userId,
            reportId: savedReport.id,
            ruleId: issue.ruleId,
            category: issue.category,
            severity: issue.severity,
            description: issue.description,
            accountRef: issue.accountRef, // e.g., { creditorName, accountNumberLast4 }
            rawData: issue // or specific fields only
          }
        });
      }

      // 4c. Save AI explanations
      for (const explanation of analysisResult.aiExplanations) {
        await db.violationExplanations.create({
          data: {
            userId,
            reportId: savedReport.id,
            violationRuleId: explanation.ruleId,
            issueKey: explanation.issueKey,
            shortSummary: explanation.shortSummary,
            detailedExplanation: explanation.detailedExplanation,
            statutesCited: explanation.statutesCited,
            model: explanation.model
          }
        });
      }

      // 5️⃣ Respond with some useful info
      res.json({
        ok: true,
        reportId: savedReport.id,
        issuesCount: analysisResult.issues.length
      });
    } catch (err: any) {
      console.error('Error in /api/credit-reports/upload:', err);
      res.status(500).json({
        error: err.message || 'Failed to upload & analyze credit report.'
      });
    }
  }
);

export default router;

3️⃣ Add a minimal HTML→PDF stub

Create a small helper that takes HTML/long text, and makes a PDF buffer using pdfkit. It won’t respect fancy HTML layout, but it gives you a real downloadable PDF immediately.

// services/htmlToPdf.ts
import PDFDocument from 'pdfkit';
import { PassThrough } from 'stream';

/**
 * Minimal HTML→PDF stub:
 * - strips tags
 * - dumps the text into a single-page PDF
 * Later you can swap this with a proper HTML renderer (Puppeteer, API, etc.).
 */
export async function htmlToPdfBuffer(html: string): Promise<Buffer> {
  const doc = new PDFDocument({ margin: 50 });
  const stream = new PassThrough();
  const chunks: Buffer[] = [];

  doc.pipe(stream);

  // Very naive "HTML" handling: strip tags
  const text = html.replace(/<[^>]+>/g, '');

  doc.fontSize(11);
  doc.text(text, {
    align: 'left'
  });

  doc.end();

  return await new Promise<Buffer>((resolve, reject) => {
    stream.on('data', (chunk) => chunks.push(chunk as Buffer));
    stream.on('end', () => resolve(Buffer.concat(chunks)));
    stream.on('error', reject);
  });
}

Route implementation
// routes/disputeLetters.ts
import { Router, Request, Response } from 'express';
import path from 'path';
import fs from 'fs/promises';
import { generateAllDisputeLetters } from '../services/enhancedDisputeLetters';
import { htmlToPdfBuffer } from '../services/htmlToPdf';
import { requireAuth, requireActiveSubscription } from '../middleware/auth';
// import db from '../db';

const router = Router();

type RecipientType = 'BUREAU' | 'COLLECTOR';

const BUREAU_CODES = {
  EXPERIAN: 'EXPERIAN',
  EQUIFAX: 'EQUIFAX',
  TRANSUNION: 'TRANSUNION'
} as const;

router.post(
  '/generate',
  requireAuth,
  requireActiveSubscription,
  async (req: Request, res: Response) => {
    try {
      const userId = req.user!.id;
      const {
        reportId,
        violationIds,
        round = 1,
        recipientType,
        target // for collector: maybe collector name like 'LVNV Funding'
      }: {
        reportId: string;
        violationIds: string[];
        round?: 1 | 2 | 3;
        recipientType: RecipientType;
        target?: string;
      } = req.body;

      if (!reportId || !Array.isArray(violationIds) || !violationIds.length) {
        return res.status(400).json({
          error: 'reportId and violationIds[] are required.'
        });
      }

      if (!['BUREAU', 'COLLECTOR'].includes(recipientType)) {
        return res.status(400).json({
          error: 'recipientType must be BUREAU or COLLECTOR.'
        });
      }

      // 1️⃣ Load user, report, issues, explanations from DB
      const user = await db.users.findUnique({
        where: { id: userId }
      });

      if (!user) {
        return res.status(404).json({ error: 'User not found.' });
      }

      const reportRecord = await db.creditReports.findUnique({
        where: { id: reportId }
      });

      if (!reportRecord) {
        return res.status(404).json({ error: 'Credit report not found.' });
      }

      const parsedReport = reportRecord.parsedJson as any; // type: CreditReport

      const issues = await db.violations.findMany({
        where: {
          id: { in: violationIds },
          userId,
          reportId
        }
      });

      if (!issues.length) {
        return res.status(404).json({
          error: 'No matching violations found for this user/report.'
        });
      }

      const explanations = await db.violationExplanations.findMany({
        where: {
          userId,
          reportId,
          violationRuleId: { in: issues.map((i: any) => i.ruleId) }
        }
      });

      // 2️⃣ Build userInfo for the letter generator
      const userInfo = {
        firstName: user.firstName || '',
        lastName: user.lastName || '',
        addressLine1: user.addressLine1 || '175 Desoto Blvd S', // default from your info if you want
        city: user.city || 'Naples',
        state: user.state || 'FL',
        zip: user.zip || '34117',
        phone: user.phone || '239-888-7959',
        email: user.email || 'jarredstarman@gmail.com',
        dob: user.dob, // if you store it
        ssnLast4: user.ssnLast4 // if you store it
      };

      // 3️⃣ Call your enhanced AI letter generator
      const { bureauLetters, collectorLetters } = await generateAllDisputeLetters({
        report: parsedReport,
        issues,
        aiExplanations: explanations,
        userInfo,
        round
      });

      let letterLabel: string;
      let html: string;

      if (recipientType === 'BUREAU') {
        const bureau = (reportRecord.bureau || 'TRI').toUpperCase();

        if (!Object.keys(BUREAU_CODES).includes(bureau)) {
          return res.status(400).json({
            error: `Unknown bureau on report: ${bureau}.`
          });
        }

        const letter = bureauLetters[bureau as keyof typeof bureauLetters];
        if (!letter) {
          return res.status(404).json({
            error: `No bureau letter generated for ${bureau}.`
          });
        }

        letterLabel = bureau;
        html = letter.html || `<pre>${letter.plainText}</pre>`;
      } else {
        // COLLECTOR
        if (!target) {
          return res.status(400).json({
            error: 'target collector name is required when recipientType=COLLECTOR.'
          });
        }

        const collectorKey = Object.keys(collectorLetters).find(
          (name) =>
            name.toLowerCase() === target.toLowerCase()
        );

        if (!collectorKey) {
          return res.status(404).json({
            error: `No collector letter generated for ${target}.`
          });
        }

        const letter = collectorLetters[collectorKey];
        letterLabel = collectorKey;
        html = letter.html || `<pre>${letter.plainText}</pre>`;
      }

      // 4️⃣ Convert HTML → PDF buffer
      const pdfBuffer = await htmlToPdfBuffer(html);

      // 5️⃣ Save to disk in /files/letters
      const lettersDir = path.join(process.cwd(), 'files', 'letters');
      await fs.mkdir(lettersDir, { recursive: true });

      const safeLabel = letterLabel.replace(/[^a-z0-9]+/gi, '_').toLowerCase();
      const fileName = `${safeLabel}_round${round}_${Date.now()}.pdf`;
      const filePath = path.join(lettersDir, fileName);

      await fs.writeFile(filePath, pdfBuffer);

      const pdfUrl = `/files/letters/${fileName}`;

      // 6️⃣ Save a record in your letters table (adapt field names)
      const savedLetter = await db.disputeLetters.create({
        data: {
          userId,
          reportId,
          recipientType,
          recipientLabel: letterLabel,
          round,
          violationIds,
          pdfPath: filePath,
          pdfUrl
        }
      });

      // 7️⃣ Respond with metadata + URL
      res.json({
        ok: true,
        letterId: savedLetter.id,
        pdfUrl,
        recipientType,
        recipient: letterLabel,
        round
      });
    } catch (err: any) {
      console.error('Error in /api/dispute-letters/generate:', err);
      res.status(500).json({
        error: err.message || 'Failed to generate dispute letter.'
      });
    }
  }
);

export default router;
