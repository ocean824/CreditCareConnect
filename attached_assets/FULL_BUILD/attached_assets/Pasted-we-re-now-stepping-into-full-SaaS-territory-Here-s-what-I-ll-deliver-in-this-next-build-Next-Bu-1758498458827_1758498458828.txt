async def paypal_ipn(request: Request):
    form_data = await request.form()
    if form_data.get("payment_status") == "Completed":
        email = form_data.get("custom")
        user = load_user(email)
        user["plan"] = "pro"
        user["upgraded"] = str(datetime.now())

🧾 3. Auto-Invoice Generator
📄 invoice_utils.py
from fpdf import FPDF
import os
from datetime import date

def generate_invoice(email, amount):
    pdf = FPDF()
    pdf.add_page()
    pdf.set_font("Arial", size=12)
    pdf.cell(200, 10, txt="Payment Receipt", ln=True, align="C")
    pdf.ln(10)
    pdf.cell(200, 10, txt=f"Customer: {email}", ln=True)
    pdf.cell(200, 10, txt=f"Amount Paid: ${amount:.2f}", ln=True)
    pdf.cell(200, 10, txt=f"Date: {date.today()}", ln=True)

    os.makedirs("invoices", exist_ok=True)
    filepath = f"invoices/{email.replace('@','_')}_{date.today()}.pdf"
    pdf.output(filepath)
    return filepath

📄 email_utils.py (Send invoice)
def send_invoice_email(email):
    filepath = generate_invoice(email, 149.99)
    with open(filepath, "rb") as f:
        attachment = f.read()

    message = MIMEMultipart()
    message["Subject"] = "Credit Repair PRO – Receipt"
    message["From"] = os.getenv("GMAIL_EMAIL")
    message["To"] = email

    message.attach(MIMEText("Thank you for upgrading to PRO. Receipt attached."))
    part = MIMEApplication(attachment, Name=os.path.basename(filepath))
    part['Content-Disposition'] = f'attachment; filename="{os.path.basename(filepath)}"'
    message.attach(part)

    with smtplib.SMTP_SSL("smtp.gmail.com", 465) as server:
        server.login(os.getenv("GMAIL_EMAIL"), os.getenv("GMAIL_APP_PASSWORD"))
        server.sendmail(message["From"], [email], message.as_string())
