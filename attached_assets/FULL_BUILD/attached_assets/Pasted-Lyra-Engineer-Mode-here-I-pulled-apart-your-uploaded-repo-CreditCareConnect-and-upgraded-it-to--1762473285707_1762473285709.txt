import type { Request, Response, NextFunction } from "express";
import { randomUUID } from "crypto";

declare module "http" {
  interface IncomingMessage {
    requestId?: string;
  }
}

export function requestId(req: Request, res: Response, next: NextFunction) {
  const incoming = req.header("x-request-id");
  const id = incoming && incoming.trim() !== "" ? incoming : randomUUID();
  req.requestId = id;
  res.locals.requestId = id;
  res.setHeader("x-request-id", id);
  next();
}

3) /server/middleware/errorHandler.ts
import type { ErrorRequestHandler } from "express";
import { AppError } from "../errors";

export const errorHandler: ErrorRequestHandler = (err, req, res, _next) => {
  const requestId = (req as any).requestId || res.locals.requestId;
  if (err instanceof AppError) {
    res.status(err.status).json({
      code: err.code,
      message: err.message,
      details: err.details,
      requestId
    });
    return;
  }

  // Unknown/unexpected
  const status = typeof err?.status === "number" ? err.status : 500;
  const code = status === 404 ? "NOT_FOUND" : "INTERNAL_ERROR";
  const message = status === 404 ? "Not Found" : "Internal Server Error";
  res.status(status).json({ code, message, requestId });
};

4) /server/middleware/rateLimit.ts
import type { Request, Response, NextFunction } from "express";
import { RateLimitError } from "../errors";

// Simple in-memory sliding window per key
type Bucket = { resetAt: number; count: number };
const buckets = new Map<string, Bucket>();

export function rateLimit({ windowMs, max, key = (req: Request) => req.ip! }: {
  windowMs: number; max: number; key?: (req: Request) => string;
}) {
  return (req: Request, _res: Response, next: NextFunction) => {
    const k = key(req);
    const now = Date.now();
    const b = buckets.get(k);
    if (!b || b.resetAt <= now) {
      buckets.set(k, { resetAt: now + windowMs, count: 1 });
      return next();
    }
    if (b.count >= max) {
      return next(new RateLimitError());
    }
    b.count++;
    return next();
  };
}

5) /server/middleware/testAuth.ts
import type { RequestHandler } from "express";

/**
 * Test-only auth shim. Enabled iff TEST_AUTH=1.
 * Use headers:
 *  - X-Test-User: email
 *  - X-Test-Role: user|admin
 */
export const testAuth: RequestHandler = (req, _res, next) => {
  if (process.env.TEST_AUTH !== "1") return next();
  const email = req.header("X-Test-User");
  const role = (req.header("X-Test-Role") || "user") as "user"|"admin";
  if (email) {
    (req as any).isAuthenticated = () => true;
    (req as any).user = { claims: { sub: `test_${email}`, email }, role };
  }
  next();
};

  app.get("/readyz", async (_req, res) => {

      await db.execute(sql`select 1`);

    } catch {
      res.status(503).send("not-ready");
    }
  });
}

/** Serve OpenAPI JSON & ReDoc */
export function registerOpenApi(app: Express) {
  const specPath = path.resolve(import.meta.dirname, "openapi.yml");
  const raw = fs.readFileSync(specPath, "utf8");
  const doc = YAML.parse(raw);

  app.get("/api/json", (_req, res) => {
    res.json(doc);
  });

  app.get("/api/docs", (_req, res) => {
    res.setHeader("content-type", "text/html; charset=utf-8");

}

import { db } from "../server/db";

async function main() {
  const admins = (process.env.ADMIN_EMAILS || "").split(",").map(s => s.trim()).filter(Boolean);
  for (const email of admins) {
    const [existing] = await db.select().from(users).where(eq(users.email, email));
    if (!existing) {
      await db.insert(users).values({
        email,
        firstName: "Admin",
        lastName: "User",
        role: "admin",
        plan: "DIY"
      });
      console.log(`Seeded admin user: ${email}`);
    } else if (existing.role !== "admin") {
      await db.update(users).set({ role: "admin" }).where(eq(users.id, existing.id));
      console.log(`Promoted to admin: ${email}`);
    } else {
      console.log(`Admin already present: ${email}`);
    }
  }
  console.log("✅ Seed finished");
}

main().catch((e) => {
  console.error(e);
  process.exit(1);
});

# CreditCareConnect – PRD MASTER

## 1) Product Overview & Scope
CreditCareConnect is an AI-assisted credit-repair platform that ingests consumer credit reports, detects violations, drafts dispute letters, mails them via Click2Mail, tracks responses, and manages upgrades via PayPal. Audience: consumers (DIY), done-for-you customers, and admins. Scope includes intake, AI analysis, letter generation, mailing/tracking, notifications, admin operations, and payments.

## 2) Personas
- **Consumer (DIY):** Uploads reports, wants guided AI help and ready-to-mail letters.
- **Managed Client (DFY):** Minimal effort; expects progress tracking and notifications.
- **Admin (Staff):** Oversees cases, bulk reminders, audit logs, mail batches, response intake.

## 3) User Journeys
1. **DIY Onboarding:** Login → Upload report → AI detects violations → Generate letters → Download or mail → Track responses.
2. **DFY Upgrade:** Select plan → PayPal order → Subscription set/flag updated → DFY workflows enabled.
3. **Admin Oversight:** View stats/users/cases → Trigger D+34 reminders → Re-mail letters → Review client responses → Audit trail.

## 4) Non-Functional Requirements
- **Performance:** P95 < 300ms for typical API reads. AI ops may be async/background.
- **Reliability:** Graceful degradation on provider failures (OpenAI/Click2Mail/PayPal).
- **Security:** Session auth (Replit OIDC), RBAC (user/admin), zod input validation, standardized errors, no secrets in logs.
- **Compliance:** Clear audit logs for admin actions; tamper-evident IDs; requestId correlation.
- **Observability:** /healthz, /readyz, structured error JSON, API docs @ /api/docs.

## 5) Tech Stack (Replit)
- **Frontend:** React + Vite, shadcn UI, React Query, Wouter.
- **Backend:** Express (TypeScript), OpenAPI served via ReDoc, requestId, rate limit, error middleware.
- **DB:** Drizzle ORM (Postgres/Neon).
- **AI:** OpenAI adapter (summarization, violation detection, letter drafting).
- **Payments/Mailing:** PayPal client, Click2Mail service.

## 6) AI Enhancements (Global)
- Auto-summarize uploaded reports and highlight likely FCRA/Metro 2 violations.
- Suggest dispute letter content, tone, and statute references.
- Generate CFPB complaint drafts from case metadata.
- Chat assistant for “what next” guidance and follow-up timing.

## 7) Risks & Mitigations
- **Provider outage:** Fallback messaging + retry queue; show “pending” states.
- **Parsing failures:** Allow manual overrides and attachments; keep raw upload for audit.
- **False positives:** Present AI suggestions as drafts; require user/admin confirmation.

## 8) Release Plan
- **R1:** Upload → AI analysis → Letter draft → Download/Mail → Track.
- **R2:** Bulk admin ops (D+34), response inbox, analytics.
